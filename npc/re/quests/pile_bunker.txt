yuno,179,174,4	script	Gomer	851,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 1000) {
		mes "[Gomer]";
		mes "Voce ta carregando muita coisa. Deixa um pouco da bagagem e volta pra mim.";
		close;
	}
	if (countitem(1549)) {
		mes "[Gomer]";
		mes "E ai, como ta indo com o Pilebuncker?";
		close;
	}
	if (Class != Job_Mechanic && Class != Job_Mechanic_T && Class != Job_Baby_Mechanic) {
		mes "[Gomer]";
		mes "Eu sou um Mecanico aposentado e acho que nao tenho nenhum assunto com voce.";
		close;
	}
	mes "[Gomer]";
	mes "E ai? Qual a sensacao de estar em uma Madogear?";
	mes "Isso e algo que nao e permitido pra quem falhou em buscar seu proprio caminho.";
	next;
	mes "[Gomer]";
	mes "Eu? Me aposentei faz tempo. Porque sofri um acidente... me machuquei feio e nao vou poder pilotar de novo. Fico chateado com isso.";
	next;
	mes "[Gomer]";
	mes "Entao decidi ficar aqui forjando equipamentos pra Madogear.";
	mes "Hein? Nao to dizendo que tenho alguma reclamacao desse trabalho.";
	next;
	mes "[Gomer]";
	mes "Ser forcado a fazer algo que voce nao gosta, nao tem nada mais miseravel que isso.";
	next;
	mes "[Gomer]";
	mes "Tem gente que e feliz so em ser apoio, sabe, tipo ser o muro pra alguem se encostar.";
	next;
	mes "[Gomer]";
	mes "Isso me faz sentir bem e estou prestes a fazer um novo equipamento. Quero que voce me ajude. O que acha?";
	next;
	switch(select("Nao to muito interessado.:O que voce vai fazer?:Onde voce consegue os materiais?:Eu trouxe os materiais.")) {
	case 1:
		mes "[Gomer]";
		mes "Nao tem o que fazer";
		mes "sobre isso entao.";
		mes "Ate a proxima!";
		close;
	case 2:
		mes "[Gomer]";
		mes "Hmm, uma barra de metal enorme que pode esmagar um oponente em um golpe!";
		next;
		mes "[Gomer]";
		mes "Nao parece nada extravagante, mas eu projetei um dispositivo com mola pra atacar com grande forca.";
		next;
		mes "[Gomer]";
		mes "E escondido, entao quando voce achar uma chance, um tiro pode deixar o oponente em estado critico!";
		mes "Mal posso esperar pra ver em acao!";
		next;
		mes "[Gomer]";
		mes "Se voce imaginar o quao boa essa arma pode ser, eu vou tornar realidade.";
		next;
		mes "[Gomer]";
		mes "Me traga 200 Steel, 30 Flexible Tubes e 1 Brocca. Eu posso fazer o que acabei de descrever com esses materiais.";
		close;
	case 3:
		mes "[Gomer]";
		mes "Voce deveria conseguir steel de varios monstros como Kobolds.";
		next;
		mes "[Gomer]";
		mes "Flexible Tubes sao mais raros, dropados por Metalings perto de Einbroch e Lighthalzen.";
		next;
		mes "[Gomer]";
		mes "Por ultimo, achar uma Brocca e muito dificil, mas voce deve dar um jeito. Haha, vai la buscar!";
		close;
	case 4:
		if (countitem(999) < 200 || countitem(7325) < 30 || countitem(1415) < 1) {
			mes "[Gomer]";
			mes "Voce nao trouxe todos os materiais que eu pedi.";
			next;
			mes "[Gomer]";
			mes "Eu tenho o costume de desistir do que to fazendo e tentar fazer coisas novas quando a inspiracao bate.";
			mes "E melhor se apressar!";
			close;
		} else {
			mes "[Gomer]";
			mes "Ah, eu tava esperando. Voce nao veio na hora, entao fiquei pensando se voce tinha esquecido. Espera ai um pouquinho.";
			specialeffect EF_REPAIRWEAPON;
			next;
			mes "[Gomer]";
			mes "Ta pronto. O que achou? Nada mal, ne? Apesar de eu mesmo ter feito, tenho que dizer que ficou bem decente.";
			next;
			mes "[Gomer]";
			mes "A combinacao de peso, volume e olha so a cor e o brilho. Captura meu coracao!";
			next;
			mes "[Gomer]";
			mes "...........";
			next;
			mes "[Gomer]";
			mes "Ah, com uma pequena melhoria, poderia ser produzido em massa e entregue. Tenho certeza que posso fazer mais desses, entao vou te dar esse.";
			next;
			mes "[Gomer]";
			mes "Ah, e acontece que eu nao precisei de todo aquele Steel no final. Entao toma, leva a maior parte de volta.";
			delitem 999,50; //Steel
			delitem 7325,30; //Tube
			delitem 1415,1; //Skewer
			getitem 1549,1; //Pilebuncker
			getitem 1360,1; //Two_Handed_Axe
			close;
		}
	}
}
