	if (MaxWeight - Weight < 1000) {
		mes "Pou favou lighten seu bag.";
		close;
	}
	.@item_req = getarg(1);
	.@req_amount = getarg(2);
	if (checkquest(getarg(0)) == 1 && countitem(.@item_req) >= .@req_amount) {
		specialeffect2 EF_BLIND;
		specialeffect2 EF_BEGINSPELL;
		progressbar "ffff00",4;
		delitem .@item_req,.@req_amount;
		if (rand(2)) {
			mes "Voce tem usou "+ callfunc("F_InsertPlural",.@req_amount,getitemname(.@item_req)) +" to criar ^999900"+ getarg(3) +" Essence^000000.";
			next;
			mes "^008800O "+getarg(3)+" Essence will ultimo permanently, unless hum um problema.^000000";
			completequest getarg(0);
			specialeffect2 EF_LORD;
			close;
		}
		mes "^008888Voce tem falhou to refine o essence usando "+ callfunc("F_InsertPlural",.@req_amount,getitemname(.@item_req)) +".^000000";
		close;
	}
	if (checkquest(getarg(0)) == 2) {
		mes "Voce tem jum criou "+getarg(3)+" Essence in este area.";
		close;
	}
	mes "Alguem jum usou este area.";
	close;
}
