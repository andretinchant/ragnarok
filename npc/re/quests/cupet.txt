function	script	cute_pet_manager	{
	// getarg(0) -> .@tame_id Array
	// getarg(1) -> .@tame_amount Array
	// getarg(2) -> .@hunt_id Array
	// getarg(3) -> .@hunt_amount Array
	// getarg(4) -> .@mob_id Array
	// getarg(5) ->  Attempt Variable
	// getarg(6) -> .@tame_gets Variable
	// getarg(7) -> .@hunt_id2 Array
	// getarg(8) -> .@hunt_amount2 Array

	mes "[Cute Pet Manager]";
	mes "Ola~! Eu sou um Cute Pet Manager";
	mes "quem e in charge of public relations";
	mes "fou o Novo Upgraded Cute Pet system.";
	next;
	
	mes "[Cute Pet Manager]";
	mes "Se you have any concerns regarding";
	mes "o Cute Pet system Eu sou aqui to guide you.";
	mes "Deixe-me saber which taming";
	mes "item you querer,";
	mes "entao Eu vou tell you o monstro";
	mes "esse voce pode tame com o item.";
	next;
	
	mes "[Cute Pet Manager]";
	mes "Tambem Eu vou explain what materiais";
	mes "voce precisum to trazer in order to";
	mes "pegar o taming item.";
	next;
	
	mes "[Cute Pet Manager]";
	mes "Se you jum trouxe materiais";
	mes "to exchange fou o taming item,";
	mes "Eu consigo exchange it fou taming itens immediately.";
	next;
	
	mes "[Cute Pet Manager]";
	mes "Porem, hum um limit";
	mes "fou exchanging taming itens.";
	mes "Voce pode apenas pegar taming itens";
	mes getarg(6) + " times. So please make sure";
	mes "quantos times you have exchanged.";
	next;
	
	mes "[Cute Pet Manager]";
	mes "So, which taming item voce querer?";
	mes "Pou favou choose one de o list.";
	next;
	
	// Create Menu System
	set .@size, getarraysize(getarg(0));
	for (set .@a, 0; .@a < .@size; set .@a, .@a + 1)
		set .@menu$, .@menu$ + (.@menu$ == "" ? "" : ":") + getitemname(getelementofarray(getarg(0), .@a));
	
	// Query Player Choice
	set .@choice, select(.@menu$) - 1;

	// Store Variables (Less Lookup)
	set .@tame_id, getelementofarray(getarg(0), .@choice);
	set .@tame_amount, getelementofarray(getarg(1), .@choice);
	set .@hunt_id, getelementofarray(getarg(2), .@choice);
	set .@hunt_amount, getelementofarray(getarg(3), .@choice);
	set .@mob_id, getelementofarray(getarg(4), .@choice);
	set .@hunt_id2, getelementofarray(getarg(7), .@choice);
	set .@hunt_amount2, getelementofarray(getarg(8), .@choice);

	// dispbottom "Tame ID: " + .@tame_id;
	// dispbottom "Tame Amount: " + .@tame_amount;
	// dispbottom "Hunt ID: " + .@hunt_id;
	// dispbottom "Hunt Name: " + getitemname(.@hunt_id);
	// dispbottom "Hunt Amount: " + .@hunt_amount;
	// dispbottom "Mob ID: " + .@mob_id;
	if (.@hunt_id2) { 
		// dispbottom "Hunt ID2: " + .@hunt_id2;
		// dispbottom "Hunt Amount2: " + .@hunt_amount2;
		set .@hunt2_count, countitem(.@hunt_id2);
	}
	if (countitem(.@hunt_id) >= .@hunt_amount && countitem(6083) > 0 && .@hunt2_count >= .@hunt_amount2) {
		mes "[Cute Pet Manager]";
		mes "Ah, you have gathered todos itens";
		mes "com seu amigos. Voce pode agora";
		mes "exchange fou um " + getitemname(.@tame_id) + ".";
		mes "Do voce quer to exchange com me?";
		next;
		if  (select("Exchange:Nao Exchange") == 2) { 
			mes "[Cute Pet Manager]";
			mes "You nao? Voce vai volte novamente.";
			close;
		}
		delitem .@hunt_id, .@hunt_amount;
		delitem 6083, 1;
		if (.@hunt_id2)
			delitem .@hunt_id2, .@hunt_amount2;
		set getarg(5), getarg(5) + 1;
		getitem .@tame_id, .@tame_amount;
		
		mes "[Cute Pet Manager]";
		mes "Wise choice.";
		mes "I esperancum you e seu pet pegar ao longo.";
		close;
	} else {
		mes "[Cute Pet Manager]";
		mes "Voce tem chosen " + getitemname(.@tame_id) + ".";
		mes "Voce pode usar it fou taming";
		mes "^FF0000" + getmonsterinfo(.@mob_id, MOB_NAME) + "^000000 monstros.";
		next;
		mes "[Cute Pet Manager]";
		mes "Trazer ^FF0000" + .@hunt_amount + " " + getitemname(.@hunt_id) + "^000000" + (.@hunt_id2 ? ", ^FF0000" + .@hunt_amount2 + " " + getitemname(.@hunt_id2) : "") + " ^000000 e ^FF00001 Dolly Capsule^000000";
		mes "entao voce pode exchange them";
		mes "fou um " + getitemname(.@tame_id) + ".";
		close;
	}
}
