c_tower1,232,222,5	script	Belljamin Button#ct	968,{
	mes "[Belljamin Button]";
	mes "O monstros aqui are capaz de control tempo.";
	if( BaseLevel < 150 ){
		mes "Volte when voce tem reached ^0000FFLevel 150^000000.";
		close;
	}
	mes "Tempo, memorias, environments... todos cum se tornar warped aqui.";
	next;
	mes "[Belljamin Button]";
	mes "Eu sou begging you nao to ask mais sobre it. It might colocar seu vidum in perigo.";
	next;
	.@menu = select("Nao ask:Ir to 2nd floou of warped torre:Ir to 3rd floou of warped torre");
	switch(.@menu){
		case 1:
			break;
		case 2:
		case 3:
			if( .@menu == 2 ){
				.@map$ = "c_tower2_";
				.@x = 268;
				.@y = 26;
			}else{
				.@map$ = "c_tower3_";
				.@x = 65;
				.@y = 147;
			}
			mes "[Belljamin Button]";
			mes "Tenhum cuidado com seu choice.";
			next;
			mes "[Belljamin Button]";
			mes "Eu precientao de ^0000FF10,000z^000000 to let you in. E um pretty perigoentao lugar to be.";
			next;
			switch( select( "Quit:Pagar 10,000z:Pou que voce precisum dinheiro?" ) ){
				case 1:
					break;
				case 2:
					if( Zeny < 10000 ){
						mes "[Belljamin Button]";
						mes "Realmente? Voce nao have o dinheiro e perguntou me to pegar you la?";
						close;
					}
					Zeny -= 10000;
					mes "[Belljamin Button]";
					mes "Eu sou nao sure how voce vai manter track of tempo in lum mas be as cuidadoentao as you can.";
					close2;
					warp .@map$, .@x, .@y;
					end;
				case 3:
					mes "[Belljamin Button]";
					mes "Eu sou aqui todos dium fou o segurancum of aventureiros. I sempre alert them sobre o perigo. That's why Eu precientao de alguns form of pagamento.";
					emotion ET_SWEAT;
					close;
			}
			break;
	}
	mes "[Belljamin Button]";
	mes "Phew...Voce tem made um bom choice. Este lugar is brutal entao voce deverium provavelmente pegar back to um seguro area.";
	close;
}
