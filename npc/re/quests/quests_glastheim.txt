glast_01,195,273,3	script	Hugin's follower#1	2_F_MAGICMASTER,{
	mes "[Hugin's Follower]";
	mes "Eu nao consigo mover entre tempo e dimensions like master Hugin mas Eu consigo pretend to.";
	next;
	mes "[Hugin's Follower]";
	mes "O que e causing este Glast Heim to be imbued com such um poderoentao escuro force?";
	next;
	if( select( "Eu nao care:Esse is interesting, pou favou me digum mais" ) == 2 ){
		mes "[Hugin's Follower]";
		mes "Obrigado to Hugin, I encontrou o caminho to access este Velho Glast Heim, which is imbued by such poderoentao escuro forces.";
		next;
		mes "[Hugin's Follower]";
		mes "O que you dizer? Do voce quer to explorar la?";
		next;
		if( select( "No exploration fou me:I realmente querer to explorar!" ) == 2 ){
			mes "[Hugin's Follower]";
			mes "Sure, mas este is nao um gratis service! Eu tenho to pegar algo in retornar. Hoho...";
			next;
			mes "[Hugin's Follower]";
			mes "Eu consigo send you to two places. Which lugar voce gostarium de like me to send you?";
			next;
			switch( select( "Cancel:Monastery Cemetery (Lv100~130) - 5000z:2nd floou Corridou (Over Lv140) - 10000z" ) ){
				case 1:
					mes "[Hugin's Follower]";
					mes "Alright, pou favou volte whenever you mudar seu mente. Eu acho you will have lots to conversar com me sobre...";
					close;
				case 2:
					.@map$ = "gl_chyard_";
					.@x = 148;
					.@y = 144;
					.@zeny = 5000;
					break;
				case 3:
					.@map$ = "gl_cas02_";
					.@x = 104;
					.@y = 31;
					.@zeny = 10000;
					break;
			}
			if( Zeny < .@zeny ){
				mes "[Hugin's Follower]";
				mes "What? Voce expect este service fou gratis?";
				close;
			}
			Zeny -= .@zeny;
			warp .@map$, .@x, .@y;
			end;
		}
	}
	mes "[Hugin's Follower]";
	mes "Bem, se you are nao interessado in it, nuncum mente.";
	close;
}
