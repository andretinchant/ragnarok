que_ng,21,72,6	script	Master Armor Craftsman	4_M_RASWORD,{
	mes "[Gyo]";
	mes "How voce gostarium de define victory?";
	mes "To fell o inimigo...";
	mes "Porem, o mais importante coisum is...";
	mes "Nuncum to fall yourself!";
	next;
	mes "[Gyo]";
	mes "Fou that, voce deve better proteger yourself.";
	mes "You nao pode proteger 'others' while you hinder seu 'own' protection.";
	next;
	if(select("Combine armadura.:End conversation.") == 2) {
		mes "[Gyo]";
		mes "Prudence is tambem um importante virtue in batalha.";
		close;
	}
	switch(select("Wolf Armguard[1]:Crescent Armguard[1]:Ninjum Scale Armadura[1]:Tenebris Latitantes[1]:Quit.")) {
	case 1:
		mes "[Gyo]";
		mes "Wolf Armguard... One beastly piece of armadura.";
		next;
		callsub L_Select,2172;
		if (countitem(2171) > 0 && countitem(6252) > 9) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 6252,10; //Wolf_Blood
			getitem 2172,1; //Wolf_Armguard
			mes "[Gyo]";
			mes "Voce feel o nature of o wolf?";
			mes "Both wild e charming...";
			close;
		}
		break;
	case 2:
		mes "[Gyo]";
		mes "Crescent Armguard... A coisum of beauty e forca... With none o lesser.";
		next;
		callsub L_Select,2173;
		if (countitem(2171) > 0 && countitem(7321) > 99) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 7321,100; //Fragment_Of_Crystal
			getitem 2173,1; //Crescent_Armguard
			mes "[Gyo]";
			mes "Such beauty...";
			mes "As se o whole corpo is overflowing com magia.";
			close;
		}
		break;
	case 3:
		mes "[Gyo]";
		mes "Ninjum Scale Armadura... E disse esse um legendary Ninjum umum vez wore it.";
		next;
		callsub L_Select,15054;
		if (countitem(7451) > 29 && countitem(7562) > 29 && countitem(6091) > 29) {  //custom translation
			delitem 7451,30; //Scale_Of_Red_Dragon
			delitem 7562,30; //Ice_Scale
			delitem 6091,30; //Dark_Red_Scale
			getitem 15054,1; //Ninja_Scale_Armor
			mes "[Gyo]";
			mes "Este is truly um legendary armadura...";
			mes "Even se Eu nao acreditar in legends...";
			close;
		}
		break;
	case 4:
		mes "[Gyo]";
		mes "Shadow Rei's Armadura... Erum made to proteger others.";
		next;
		callsub L_Select,15055;
		if (countitem(15056) > 0 && countitem(6089) > 9) {  //custom translation
			delitem 15056,1; //Special_Ninja_Suit_
			delitem 6089,10; //Piece_Of_Darkness
			getitem 15055,1; //Tenebris_Latitantes
			mes "[Gyo]";
			mes "Como estum it?";
			mes "Does o armadurum nao aparecer as se hum two?";
			close;
		}
		break;
	case 5:
		close;
	}
	mes "[Gyo]";
	mes "Voce nao have suficiente ingredientes.";
	mes "Olhar novamente...";
	close;

L_Select:
	.@string1$ = "Check Ingredients.";
	.@string2$ = "Check Stats.";
	while(1) {
		switch(select(.@string1$,.@string2$,"Combine.","Quit.")) {
		case 1:
			.@string1$ = "";
			callsub L_Ingredients, getarg(0);
			break;
		case 2:
			.@string2$ = "";
			callsub L_Stats, getarg(0);
			break;
		case 3:
			callsub L_Combine;
			return;
		case 4:
			mes "[Gyo]";
			mes "Looks like voce precisum mais tempo to decide.";
			close;
		}
	}

L_Ingredients:
	mes "[Gyo]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "To fazer um ^ff0000Wolf Armguard, voce precisum 1 Fox Armguard e 10 Blood of Wolf^000000.";
		mes "Voce pode feel it? O throbbing of o wolf sangue...";
		break;
	case 2173:	//Crescent_Armguard
		mes "To fazer um ^ff0000Crescent Armguard, voce precisum 1 Fox Armguard e 100 Crystal Fragments^000000.";
		mes "A crescent as bonito e clear as um crystal sera engraved on it.";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "To fazer um ^ff0000Ninjum Scale Armadura, voce precisum 30 Fogo Dragon Scales, 30 Ice Scales, e 30 Darkred Scale Pieces^000000.";
		mes "Voce pode feel seu legendary powers.";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "To fazer um ^ff0000Shadow Rei's Armadura, voce vai precisar um Especial Ninjum Suit com um slot e 10 Escuro Pieces^000000.";
		mes "With o escuridao it erases existence itself...";
		break;
	}
	next;
	return;

L_Stats:
	mes "[" + getitemname(getarg(0)) + "]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "Chance to have 5 sec. of ATK+100, FLEE -50 upon melee atacar.";
		mes "Escudo Type, Defense 45, Slot 1";
		mes "Required Nivel 100, Ninjum Type Apenas";
		break;
	case 2173:	//Crescent_Armguard
		mes "Decreases post habilidade delay fou 2% fou cadum enhancement.";
		mes "Escudo Type, Defense 70, Slot 1";
		mes "Required Nivel 100, Ninjum Type Apenas";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "MHP+15%, MSP-30%";
		mes "Armadurum Type, Defense 90, Slot 1";
		mes "Required Nivel 100, Ninjum Type Apenas";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "Chance to activate Lvl. 1 Illusion - Shadow when attacked by melee ataques.";
		mes "Armadurum Type, Defense 60, Slot 1";
		mes "Required Nivel 100, Ninjum Type Apenas";
		break;
	}
	next;
	return;

L_Combine:
	mes "[Gyo]";
	mes "Fazer sure esse ^ff0000you have o correct ingredientes e equipamento at mao^000000.";
	mes "No usar crying sobre it later.";
	next;
	switch(select("Combine.:Quit.")) {
	case 1:
		if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
			mes "- Hold on!! -";
			mes "- You nao pode receber itens -";
			mes "- porque you carregar tambem muito. -";
			mes "- Pou favou tentar novamente -";
			mes "- depois lightening seu burden. -";
			close;
		}
		return;
	case 2:
		mes "[Gyo]";
		mes "Looks like voce precisum mais tempo to decide.";
		close;
	}
	end;
}
