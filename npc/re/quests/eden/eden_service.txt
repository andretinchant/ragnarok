-	script	Goods Cabinet#00::pggc	-1,{
	mes "Isentao e um Goods Armazem Armario.";
	mes "A message is written on um piece of paper.";
	next;
	mes "+ Armario is exclusively +";
	mes "+ fou o Eden grupo +";
	mes "Se voce quer to usar this";
	mes "armario pou favou verificar";
	mes "o following:";
	next;
	mes "1.Voce e um membro";
	mes "   of Eden grupo?";
	mes "2.Voce jum aprendeu";
	mes "   suficiente basic habilidades?";
	mes "3.Armario taxum is";
	mes "  ^4d4dff500 zeny^000000!";
	next;
	if (callfunc("F_HasEdenGroupMark")) {
		mes "Voce precisum to insert zeny to usar o armario.";
		mes "Custo : 500 Zeny ";
		mes "Voce gostarium de like to usar it?";
		next;
		switch(select("Usar o Armario.:Cancelar.")) {
			case 1:
				if (Zeny > 499) {
					if(!callfunc("F_CanOpenStorage")){
						mes "O armario is nao funcionando fou me.";
						mes "Talvez Eu sou nao yet qualificado to usar Armario.";
						close;
					}
					set Zeny, Zeny - 500;
					close2;
					openstorage;
					end;
				}
				mes "Eu nao have suficiente zeny.";
				mes "Eu precientao de at menos 500 zeny to usar o Armario.";
				close;
			case 2:
				mes "Eu vou usar it proximo tempo.";
				close;
		}
	}
	mes "Eu precientao de um Eden Grupo Mark to usar este Armario.";
	close;
}
