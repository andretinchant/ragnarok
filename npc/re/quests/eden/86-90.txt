L_Quest:
	set .@quest1, getarg(0);
	set .@quest2, .@quest1+31;
	.@cutin$ = getarg(3);
	set .@playtime, checkquest(.@quest2,PLAYTIME);
	if (.@playtime == 0 || .@playtime == 1) {
		mes "[86-90 Mission Board]";
		mes "Voce precisum to esperar 3 hours antes voce pode pegar este missao novamente.";
		close;
	}
	set .@hunting, checkquest(.@quest1,HUNTING);
	if (.@hunting == 0 || .@hunting == 1) {
		mes "[86-90 Mission Board]";
		mes "Voce jum terminou cacando todos 30 "+getarg(1)+getarg(5,"")+" yet?";
		close;
	} else if (.@hunting == 2) {
		mes "[86-90 Mission Board]";
		mes "Voce tem completou o cacando.";
		mes "Pou favou aceitar este recompensum as um compensation.";
		erasequest .@quest1;
		if (.@playtime > -1) erasequest .@quest2;
		setquest .@quest2;
		getexp getarg(2),0;
		close;
	}/* else if (.@hunting == 3) {
//		recall_completequest .@quest1

		mes "[86-90 Mission Board]";
		mes "Voce tem re-aceitou o "+getarg(1)+" Cacadum missao.";
		close;
	}*/ else {

		mes getarg(4);

		mes "[86-90 Mission Board]";
		mes "Cacar 30 "+getarg(1)+" monstros lum e post seu sucesso on este quadro.";
		next;
		if (.@cutin$ != "") cutin .@cutin$,255;
		if(select("Eu vou cacar them.:No obrigado.") == 1) {
			mes "[86-90 Mission Board]";
			mes "When you have completou o missao, post it on este quadro e coletar seu recompensa.";
			if (.@playtime > -1) erasequest .@quest2;
			setquest .@quest1;
		}
