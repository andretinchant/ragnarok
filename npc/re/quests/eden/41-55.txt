moc_para01,40,38,3	script	Mission [41 - 55]	857,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "Voce nao e membro do grupo Eden. Voce nao esta qualificado para acessar o quadro de avisos.";
		close;
	}
	if (checkweight(1201,1) == 0 || (MaxWeight - Weight) < 1000) {
		mes "^008800Espere um pouco!!";
		mes "You nao pode receber any mais itens porque voce e carregando tambem muitos itens. Pou favou tentar novamente depois you lighten up.^000000";
		close;
	}
	if (checkquest(12088) == -1) {
		if (BaseLevel < 41 || BaseLevel > 55) {
			mes "Missao quadro de avisos fou Lv41~55 aventureiros.";
			close;
		}
		mes "[Mission Board]";
		mes "Quadro de missoes.";
		next;
		if(select("Ver as Missoes.:Cancelar.") == 2) {
			mes "^000077Ok. Vamos verificar o missoes proximo tempo.^000000.";
			close;
		}
		switch(select("Tharum Frog Cacada:Cruiser Cacada:Kukre Cacada:Orc Baby Cacada:Orc Guerreiro Cacada:Vadon Cacada:Megalodon Cacada:Marse Cacada:Cornutus Cacada:Myst Cacada:Shellfish Cacada:Marine Sphere Cacada")) {
			case 1: callsub L_Quest,12072,"Voce jum visto um red frog? Isentao e chamado Tharum Frog e e muito rare. Nos estamos tentando to verify where it inhabits. Entao pou favou cacar 20 Tharum Frogs.";
			case 2: callsub L_Quest,12073,"Eu ouvi of alguns toys atacando pessoas. Isentao e ridiculous! Vamos show them nosso poder! Nos temos to cacar 15 Cruisers!";
			case 3: callsub L_Quest,12074,"Alguem got roubado perto o praia! O possivel suspeito is um criminoso monstro chamado Kukre. Pou favou cacar 30 Kukre.";
			case 4: callsub L_Quest,12075,"Orc Babies are ficando mais violent estes days. Pou favou cacar 15 Orc Babies.";
			case 5: callsub L_Quest,12076,"Orc Guerreiros are aumentando o aggression. Pou favou cacar 20 Orc Guerreiros.";
			case 6: callsub L_Quest,12077,"Voce jum visto crabs com thick shells? Eles sao Vadons. Pou favou cacar 15 Vadons.";
			case 7: callsub L_Quest,12078,"Morto fish are todos ao redou o praium e it is ficando worse. Pou favou cacar 30 Megalodon.";
			case 8: callsub L_Quest,12079,"Voce pode cacar Marse? Eu precientao de 15 Marse fou meu study. Pou favou me ajude.";
			case 9: callsub L_Quest,12080,"Um idiot Cornutus bit o Prefeito's pe. O Mayou is chateado entao nos precisamos to cacar 20 Cornutus to fazer him feliz!";
			case 10: callsub L_Quest,12081,"Pessoas are freaking out porque of Myst. It looks like um ghost so, it scares pessoas. Pou favou cacar 15 Myst.";
			case 11: callsub L_Quest,12082,"Nao ask why, mas pou favou cacar 30 Shellfish. Isentao e um segredo missao!";
			case 12: callsub L_Quest,12083,"Marine Sphere? Do voce sabe what um Marine Sphere is? Eu tenho um missao fou you to cacar 15 Marine Sphere, entao pou favou do so.";
			case 13: callsub L_Quest,12084,"O mayou wants to eat um delicioso fish dish. Phen sera o perfect ingrediente fou it. Pou favou cacar 20 Phen!";
			case 14: callsub L_Quest,12085,"Morto Orc monstros got back to nosso cidade e they've revived as Orc Skeletons. Nos estamos in caos! Pou favou cacar 15 Orc Skeleton.";
			case 15: callsub L_Quest,12086,"Alguns estranho monstros chamado Zenorc are atacando us. Pou favou cacar 30 Zenorc.";
			case 16: callsub L_Quest,12087,"Estejum preparado! Isentao e um cruel missao! Cacar 15 Mummies!! I tentou umum vez, mas I nao conseguiu conseguir. Isentao e tambem assustador!";
		}
		end;
	}
	setarray .@Exp[0],9600,7200,15750,7875,11100,8550,17100,9000,12300,9900,20250,10350,13800,11025,22500,11250;
	for(set .@i,0; .@i<16; set .@i,.@i+1)
		if (checkquest(12072+.@i,HUNTING) == 2) {
			mes "[Mission Board]";
			mes "Voce completou a missao.";
			mes "Aqui esta sua recompensa.";
			erasequest 12088;
			erasequest 12072+.@i;
			specialeffect2 EF_STEAL;
			getexp .@Exp[.@i],0;
			close;
		}
	if ((checkquest(12072,HUNTING) == 0) && (checkquest(12073,HUNTING) == 0) && (checkquest(12074,HUNTING) == 0) && (checkquest(12075,HUNTING) == 0) && (checkquest(12076,HUNTING) == 0) && (checkquest(12077,HUNTING) == 0) && (checkquest(12078,HUNTING) == 0) && (checkquest(12079,HUNTING) == 0) && (checkquest(12080,HUNTING) == 0) && (checkquest(12081,HUNTING) == 0) && (checkquest(12082,HUNTING) == 0) && (checkquest(12083,HUNTING) == 0) && (checkquest(12084,HUNTING) == 0) && (checkquest(12085,HUNTING) == 0) && (checkquest(12086,HUNTING) == 0) && (checkquest(12087,HUNTING) == 0)) {
		mes "Voce pode ver o quadro de avisos agora.";
		erasequest 12088;
		close;
	}
	mes "Nao ha nada especial para fazer.";
	close;
L_Quest:
	mes "[Mission Board Memo]";
	mes getarg(1);
	next;
	if(select("Aceitar a missao.:Cancelar.") == 1) {
		mes "[Mission Board]";
		mes "Voce tem aceitou o missao. Fou mais detalhes, pou favou press 'alt + u' keys, voce pode pegar informacao de o Missao Window.";
		setquest 12088;
		setquest getarg(0);
	}
	close;
}
