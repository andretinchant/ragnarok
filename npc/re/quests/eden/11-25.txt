moc_para01,36,38,3	script	Mission [11 - 25]#Tuto	4_BOARD3,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "- Voce precisum to have um -";
		mes "- ^4d4dff'Eden Grupo Mark'^000000 -";
		mes "- to receber estes missoes. -";
		close;
	}
	for ( .@i = 11124; .@i < 11134; .@i++ )
		if (checkquest(.@i,PLAYTIME) == 2) erasequest .@i;
	for( .@i = 11114; .@i < 11124; .@i++ ) {
		.@j = checkquest(.@i,HUNTING);
		if (.@j == 2) {
			mes " - Voce deve coletar seu - ";
			mes " - recompensum antes starting - ";
			mes " - um novo missao. - ";
			mes " ";
			mes " - Manager : Spike - ";
			close;
		}
		else if (.@j >= 0) {
			mes " - Voce pode apenas request - ";
			mes " - one missao at um tempo. - ";
			mes " ";
			mes " - Manager : Spike - ";
			close;
		}
	}
	if (BaseLevel < 11) {
		mes " - Hum no tasks - ";
		mes " - abrir fou you agorum mesmo. - ";
		close;
	}
	if (BaseLevel > 25) {
		mes " - Estes missoes are tambem - ";
		mes " - facil fou seu caliber - ";
		mes " - procurar fou um mais nivel - ";
		mes " - appropriate missao. - ";
		close;
	}
	mes " - Missao quadro de avisos -";
	mes " - fou beginners. -";
	mes " ";
	mes " - Manager : Spike -";
	next;
	switch(select("Hornet Cacada:Condou Cacada:Grasshopper's Leg:Worm Tail Cacada:Spore Cacada:Pest Control:Mukum Cacada:Familiar Cacada:Coletar Feathers:Coletar Poison Spores:Cancelar")) {
		case 1: callsub L_Quest,11114,"Hornet Cacada"," - Cacar 10 ^4d4dff'Hornets'^000000 - "," - northwest of Prontera. - ";
		case 2: callsub L_Quest,11115,"Condou Cacada"," - Cacar 20 ^4d4dff'Condors'^000000 - "," - sul of Morocc. - ";
		case 3: callsub L_Quest,11116,"Grasshopper's Leg"," - Cacar 10 ^4d4dff'Rockers'^000000 - "," - southwest of Pronterum e - "," - trazer 10 ^4d4dffGrasshopper's Legs^000000. - ";
		case 4: callsub L_Quest,11117,"Worm Tail Cacada"," - Cacar 20 ^4d4dff'Worm Tails'^000000 - "," - southeast of Payon.- ";
		case 5: callsub L_Quest,11118,"Spore Cacada"," - Cacar 30 ^4d4dff'Spores'^000000 - "," - ao redou Payon. - ";
		case 6: callsub L_Quest,11119,"Pest Control"," - Cacar 20 ^4d4dffThief Bug Eggs^000000 - "," - in o Pronterum Culvert - "," - e trazer 10 ^4d4dffChrysalis^000000. - ";
		case 7: callsub L_Quest,11120,"Mukum Cacada"," - Cacar 20 ^4d4dff'Muka'^000000 - "," - sul of Prontera. - ";
		case 8: callsub L_Quest,11121,"Familiar Cacada"," - Cacar 20 ^4d4dffFamiliar^000000 - "," - in o Culvert Sewers - "," - oeste of Prontera. - ";
		case 9: callsub L_Quest,11122,"Coletar Feathers"," - Cacar 30 ^4d4dff'Peco Pecos'^000000 - "," - sul of Prontera. - ";
		case 10: callsub L_Quest,11123,"Coletar Poison Spores"," - Cacar 30 ^4d4dffPoison Spores^000000 - "," - dentro o Geffen Torre e - "," - trazer 5 ^4d4dffPoison Spores^000000. - ";
		case 11:
			mes " - Parar lendo o quadro de avisos. - ";
			close;
	}

L_Quest:
	.@quest_status = checkquest(getarg(0)+10,PLAYTIME);
	if (.@quest_status == 0 || .@quest_status == 1) {
		mes " - Missao ^4d4dff"+ getarg(1) +"^000000 - ";
		mes " - is jum terminou fou hoje. - ";
		mes " ";
		mes " - Manager : Spike - ";
		close;
	}
	mes " - Missao '"+ getarg(1) +"' - ";
	for ( .@i = 2; .@i < getargcount(); .@i++ )
		mes getarg(.@i);
	mes " ";
	mes " - Manager : Spike - ";
	next;
	if (select("Eu vou do este missao.:Olhar fou outro missoes.") == 2) {
		mes " - Vamos procurar fou -";
		mes " - outro missoes. - ";
		close;
	}
	for ( .@i = 2; .@i < getargcount(); .@i++ )
		mes getarg(.@i);
	setquest getarg(0);
	close;
}

moc_para01,32,30,6	script	Spike	4_F_CHILD,{
	for( .@i = 11114; .@i < 11124; .@i++ ) {
		.@j = checkquest(.@i,HUNTING);
		if (.@j == 2) {
			switch(.@i) {
				case 11114: callsub L_Quest,.@i,"Hornet Hunting",900;
				case 11115: callsub L_Quest,.@i,"Condor Hunting",1800;
				case 11116: callsub L_Quest,.@i,"Grasshopper's Leg",1400,940,10;
				case 11117: callsub L_Quest,.@i,"Worm Tail Hunting",2600;
				case 11118: callsub L_Quest,.@i,"Spore Hunting",3900;
				case 11119: callsub L_Quest,.@i,"Extirpate Insects",2900,915,10;
				case 11120: callsub L_Quest,.@i,"Muka Hunting",3200;
				case 11121: callsub L_Quest,.@i,"Familiar Hunting",3200;
				case 11122: callsub L_Quest,.@i,"Collect Feathers",5400;
				case 11123: callsub L_Quest,.@i,"Collect Poison Spores",6000,7033,5;
			}
		}
		else if (.@j >= 0) {
			mes "[Spike]";
			mes "You nao tenho terminou seu missao yet.";
			mes "Manter up o bom trabalhar. Voce e almost done.";
			close;
		}
	}
	mes "[Spike]";
	mes "Voce e procurando fou um job?";
	mes "We sempre have lots of trabalhar to do.";
	next;
	mes "[Spike]";
	mes "Why nao voce parece ao redou e encontrar alguns interesting tasks?";
	close;

L_Quest:
	.@quest_id = getarg(0);
	if (checkquest(.@quest_id,HUNTING) == 2) {
		.@item_req = getarg(3,0);
		.@req_amount = getarg(4,0);
		if (.@item_req) {
			if (countitem(.@item_req) < .@req_amount) {
				mes "[Spike]";
				mes "Voce nao have suficiente "+ getitemname(.@item_req) +".";
				mes "Voce precisum to trazer "+ .@req_amount +" "+ getitemname(.@item_req) +".";
				close;
			}
		}
		mes "[Spike]";
		mes "Hmm... deixe-me ver~";
		mes "Voce tem completou o '"+ getarg(1) +"' missao.";
		next;
		mes "[Spike]";
		mes "Bom trabalhar~!!";
		mes "Nosso client is entao feliz com seu trabalhar. He will appreciate what you have done.";
		mes "Pou favou ajudar us novamente. Obrigado!";
		if (.@item_req)
			delitem .@item_req,.@req_amount;
		erasequest .@quest_id;
		setquest .@quest_id+10;
		getexp getarg(2),0;
		close;
	}
}
