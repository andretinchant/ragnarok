moc_prydb1,39,129,2	script	Thief Guide#thief	69,{
	if (Upper == 1) {
		if (ADVJOB == Job_Assassin_Cross || ADVJOB == Job_Stalker) {
			if (Class == Job_Novice_High) {
				mes "[Thief Guide]";
				mes "Hein? Eu te conheco? E assustador voce parecer tao familiar. Voce nao tem um gemeo, tem?";
				next;
				if (!callfunc("F_CanChangeJob")) {
					mes "[Thief Guide]";
					mes "O que, voce quer ser um Gatuno? Desculpe, mas parece que voce precisa de mais treinamento.";
					next;
					mes "[Thief Guide]";
					mes "Nao tenha pressa e aprenda todas as Habilidades Basicas, ta? Bem, ate mais~!";
					close;
				}
				mes "[Thief Guide]";
				mes "Bem, tenho a sensacao de que voce ja passou por uma vida inteira de lutas, entao vou te promover a Gatuno agora mesmo. E melhor dar a voces, duroes, o que querem...";
				next;
				skill "NV_TRICKDEAD",0,SKILL_PERM;
				jobchange Job_Thief_High;
				skill "TF_SPRINKLESAND",1,SKILL_PERM;
				skill "TF_BACKSLIDING",1,SKILL_PERM;
				skill "TF_PICKSTONE",1,SKILL_PERM;
				skill "TF_THROWSTONE",1,SKILL_PERM;
				mes "[Thief Guide]";
				mes "Ja que voce se tornou um Gatuno, viva como um Gatuno. Agora, va em frente! Proximo~";
				close;
			}else{
				mes "[Thief Guide]";
				if (Sex == SEX_MALE)
					mes "E ai, cara.";
				else
					mes "E ai, gatinha~";
				close;
			}
		}else{
			mes "[Thief Guide]";
			if (Sex == SEX_MALE)
				mes "E ai, cara.";
			else
				mes "E ai, gatinha.";
			mes "...Ei! Voce parece certinho demais pra querer ser Gatuno!! Agora sai fora, to ocupado. Proximo!";
			close;
		}
	}
	if(q_job_thief == 1) {
		mes "[Thief Guildsman]";
		mes "Ehh... Parece "+(Sex == SEX_MALE?"filho":"filha")+" querido de familia rica, mas por que voce esta aqui nesse lugar sujo?";
		next;
		mes "[Thief Guildsman]";
		mes "Bem, eu nao sou responsavel por te tornar um Gatuno. Eu so aceito inscricoes, entendeu?";
		mes "Se voce quer ser Gatuno, pergunte ao cara de olhar afiado do meu lado.";
		close;
	}
	mes "[Thief Guildsman]";
	if(BaseJob != Job_Novice) {
		if(BaseJob == Job_Thief) {
			mes "Ei~ se tiver algum problema, pode desabafar comigo a qualquer hora, hein?";
			close;
		}else{
			mes "Que diabos...?";
			if(Sex == SEX_MALE) {
				mes "Ha... voce e um "+jobname(Class)+", cabecao!";
				next;
				mes "[Thief Guildsman]";
				mes "Ei, irmao.";
			}else{
				next;
				mes "[Thief Guildsman]";
				mes "Ei, mocinha.";
			}
			mes "Por que voce esta aqui? Volte pro seu lugar~ volte~~";
			close;
		}
	}
	mes "Ehh... Parece "+(Sex == SEX_MALE?"filho":"filha")+" querido de familia rica, mas por que voce esta aqui nesse lugar sujo?";
	next;
	if(select("Quero ser um Gatuno.:Nada.") == 2) {
		mes "[Thief Guildsman]";
		mes "Voce sabe que nao pode ser gatuno sem uma inscricao...";
		mes "O que voce tem em mente...?";
		close;
	}
	mes "[Thief Guildsman]";
	mes "Bem, voce tem tanto orgulho disso?";
	mes "Voce esta me dizendo com tanto orgulho que quer ser Gatuno! Por que nao vai a todas as vilas e se anuncia como gatuno?";
	mes "'Ha ha ha! Va colocar uma faixa que diz 'Serei um orgulhoso gatuno que rouba as coisas dos outros.'";
	next;
	mes "[Thief Guildsman]";
	mes "Voce quer tanto ser um gatuno?";
	next;
	switch(select("Sim.:Nao.:E voce?")) {
	case 1:
		mes "[Thief Guildsman]";
		mes "Oh, quer mesmo...? Ha.. bem... vivendo e aprendendo a ver pessoas estranhas como voce.";
		next;
		break;
	case 2:
		mes "[Thief Guildsman]";
		mes "Entao por que voce esta aqui? Acha que pode se tornar gatuno tao facilmente?";
		next;
		break;
	case 3:
		mes "[Thief Guildsman]";
		mes "Eh..? eu? eu?";
		mes "Bem... eu simplesmente me encaixo bem em ser gatuno... pela personalidade... nao me importo com essa bobagem.";
		next;
		break;
	}
	mes "[Thief Guildsman]";
	mes "De qualquer forma, no mundo la fora, nunca diga que quer ser Gatuno!!";
	next;
	mes "[Thief Guildsman]";
	mes "Entao, voce quer se inscrever para ser Gatuno?";
	next;
	if(select("Sim, eu quero.:Nao.") == 2) {
		mes "[Thief Guildsman]";
		mes "Bem... faca o que quiser~ siga seu caminho~";
		close;
	}
	mes "[Thief Guildsman]";
	mes "Certo, me diga seu nome. Ha? O que?";
	mes ""+strcharinfo(0)+". Entao "+strcharinfo(0)+", certo...?";
	next;
	if (!callfunc("F_CanChangeJob")) {
		mes "[Thief Guildsman]";
		mes "Posso ver sua forte vontade de se tornar Gatuno......";
		mes "Mas so com vontade, voce nao vai conseguir numa luta de verdade, nao e?";
		mes "Entao va e alcance pelo menos o Nivel 9 de Habilidade Basica.";
		close;
	}
	mes "[Thief Guildsman]";
	mes "Hmm.... encontrei seus interessantes antecedentes criminais nos dados da agencia de detetives. Bem, parece que voce tem o que e preciso para ser um bom gatuno.";
	mes ""+strcharinfo(0)+". Eu agora declaro que voce passou na entrevista de emprego!";
	next;
	mes "[Thief Guildsman]";
	mes "Bem, eu nao sou responsavel por te tornar um Gatuno. Eu so aceito inscricoes, entendeu?";
	mes "Se voce quer ser Gatuno, pergunte ao cara do meu lado.";
	set q_job_thief,1;
	close;
}
