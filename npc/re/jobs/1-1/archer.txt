payon_in02,64,71,4	script	Archer Guildsman#archer	85,{
	if (Upper == 1) {
		if (Class == Job_Novice_High && (ADVJOB == Job_Sniper || ADVJOB == Job_Clown || ADVJOB == Job_Gypsy)) {
			mes "[Archer Guildsman]";
			mes "Ei, eu te conheco.";
			mes "Voce ja fez esse teste";
			mes "antes, nao foi?";
			next;
			mes "[Archer Guildsman]";
			mes "Ah, voce deve ter ido";
			mes "a Valhalla e renascido.";
			mes "Uau, isso e muito impressionante!";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Archer Guildsman]";
				mes "Err...";
				mes "E melhor voce aprender todas as Habilidades Basicas primeiro antes de se tornar um Arqueiro.";
				next;
				mes "[Archer Guildsman]";
				mes "Certo, ate mais.";
				close;
			}
			mes "[Archer Guildsman]";
			mes "Muito bem. Eu nao";
			mes "preciso dizer mais nada.";
			mes "Sei que voce sera um otimo Arqueiro...";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Archer_high;
			skill "AC_MAKINGARROW",1,SKILL_PERM;
			skill "AC_CHARGEARROW",1,SKILL_PERM;
			mes "[Archer Guildsman]";
			mes "Embora nao haja recompensa especial para voce dessa vez, espero que entenda. Cuide-se.";
			close;
		}else{
			mes "[Archer Guildsman]";
			mes "Oh...?";
			mes "Ei, o que voce";
			mes "esta fazendo aqui...?";
			next;
			mes "[Archer Guildsman]";
			mes "Posso ver que voce nao tem vocacao para ser Arqueiro. Parece que voce foi feito para";
			mes "outra coisa...";
			close;
		}
	}
	mes "[Archer Guildsman]";
	mes "Prazer em conhece-lo. Como posso ajuda-lo?";
	next;
	switch(select("Quero ser um Arqueiro.:Preciso saber os requisitos, por favor.:Nada, obrigado.")) {
	case 1:
		mes "[Archer Guildsman]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Archer)
				mes "Haha, voce esta brincando comigo...";
			else {
				mes "Sinto muito, mas apenas Aprendizes podem mudar de classe.";
				mes "Voce ja tem sua propria classe, nao tem?";
			}
			close;
		}else{
			mes "Voce e... "+strcharinfo(0)+", certo?";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Archer Guildsman]";
				mes "Bem, voce nao esta no nivel de habilidade correto.";
				mes "Seu nivel de classe deve ser pelo menos ^4A4AFF10^000000 e seu nivel de Habilidade Basica deve chegar ao ^4A4AFFnivel 9";
				next;
				mes "[Archer Guildsman]";
				mes "Porque um Arqueiro precisa de concentracao extremamente alta, entao nao aceitamos aqueles que tem pouca paciencia.";
				close;
			}
			mes "[Archer Guildsman]";
			mes "Sua Habilidade Basica ja e suficiente..";
			mes "....Hm~~ entao voce ja esta pronto para ser um Arqueiro. Vou fazer o procedimento agora mesmo.";
			next;
			mes "[Archer Guildsman]";
			mes "Parabens! Voce agora e um Arqueiro! Alem disso, esperamos que voce participe ativamente de muitos programas para o reavivamento da Guilda dos Arqueiros.";
			mes "Ah, itens chegaram do Departamento de Producao. Aqui, pegue! Sao todos seus!";
			callfunc "Job_Change",Job_Archer;
			getitem 1742,1;	// N_Composite_Bow
			getitem 12004,1;	// Arrow_Container
			getitem 12009,1;	// Silver_Arrow_Container
			getitem 12008,1;	// Fire_Arrow_Container
			next;
			mes "[Archer Guildsman]";
			mes "Com um arco e flechas, agora voce se tornou um verdadeiro Arqueiro.";
			mes "Se voce abrir o recipiente de flechas, ha flechas dentro e entao voce pode equipa-las.";
			mes "Bem, espero ouvir boas noticias suas. Esta na hora de dizer adeus.";
			mes "Ate mais.";
			close;
		}
	case 2:
		mes "[Archer Guildsman]";
		mes "Um Arqueiro tem habilidades usando arco e possui varios talentos.";
		mes "A maior habilidade de um Arqueiro e atacar inimigos a longa distancia.";
		next;
		mes "[Archer Guildsman]";
		mes "Embora um Arqueiro tenha HP mais fraco, ele pode atingir inimigos a longa distancia,";
		mes "entao um Arqueiro e mais seguro em uma batalha real.";
		next;
		mes "[Archer Guildsman]";
		mes "Embora um Arqueiro em Ragnarok tenha HP mais baixo, ele tem alta precisao e taxa de ataque para que o arqueiro possa matar monstros antes que eles cheguem perto.";
		next;
		mes "[Archer Guildsman]";
		mes "^8C2121Um Arqueiro pode mudar de classe para Cacador.^000000";
		mes "^8C2121Alem de Cacador, se voce for homem, pode mudar para Bardo e se for mulher, pode mudar para Odalisca.^000000";
		close;
	case 3:
		mes "[Archer Guildsman]";
		mes "Se tiver alguma duvida, fique a vontade para vir me perguntar.";
		close;
	}
}
