prt_church,184,41,4	script	Cleric#aco	60,{
	if (Upper == 1) {
		if (ADVJOB == Job_High_Priest || ADVJOB == Job_Champion) {
			if (Class == Job_Novice_High) {
				mes "[Father Mareusis]";
				mes "Ah, sinto que voce suportou";
				mes "uma experiencia de vida passada. Voce deve ter aprendido muitas coisas antes de entrar em Valhalla.";
				next;
				if (!callfunc("F_CanChangeJob")) {
					mes "[Father Mareusis]";
					mes "Infelizmente, acho que voce nao esta pronto para se tornar um Acolito ainda. Por favor, termine de aprender todas as Habilidades Basicas primeiro.";
					next;
					mes "[Father Mareusis]";
					mes "Enquanto isso,";
					mes "eu esperarei ate";
					mes "que voce esteja pronto.";
					mes "Que Deus esteja";
					mes "com voce.";
					close;
				}
				mes "[Father Mareusis]";
				mes "Bem, eu te dou boas-vindas";
				mes "de volta de Valhalla e";
				mes "desejo boa sorte na sua";
				mes "nova jornada de vida.";
				next;
				skill "NV_TRICKDEAD",0,SKILL_PERM;
				jobchange Job_Acolyte_High;
				skill "AL_HOLYLIGHT",1,SKILL_PERM;
				mes "[Father Mareusis]";
				mes "Agora, aventure-se e procure aqueles que precisam da sua ajuda. Que Deus ilumine seu caminho.";
				close;
			}else{
				mes "[Father Mareusis]";
				mes "Agora, aventure-se para buscar pessoas que precisam da sua ajuda. Que Deus ilumine seu caminho.";
				close;
			}
		}else{
			mes "[Father Mareusis]";
			mes "Sinto que voce suportou uma experiencia de vida passada. Voce deve ter aprendido muitas coisas antes de entrar em Valhalla.";
			next;
			mes "[Father Mareusis]";
			mes "No entanto, posso dizer que voce nao e adequado para ser um Acolito. Por favor, lembre-se de quem voce foi na sua vida passada e encontre seu caminho.";
			close;
		}
	}
	mes "[Father Mareusis]";
	mes "O que voce procura?";
	next;
	switch(select("Mudar de classe para acolito.:Perguntar os requisitos para ser acolito.:Desistir.")) {
	case 1:
		mes "[Father Mareusis]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Acolyte)
				mes "Voce esta se sentindo bem hoje? Posso ver pela sua vestimenta que voce ja e um Acolito. Voce nao esta brincando, esta?";
			else
				mes "Desculpe, mas parece que voce ja tem sua propria classe, nao e?";
			close;
		}else{
			mes "Voce realmente deseja se tornar um servo de Deus?";
			mes "Vamos ver se voce esta pronto para isso ou nao... Hmm...";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Father Mareusis]";
				mes "Meu Deus?! Voce ainda nao completou a pratica basica?! Voce tem um longo caminho pela frente! Volte depois de aumentar seu nivel de classe!";
				close;
			}
			mes "[Father Mareusis]";
			mes "Hmm... seu nivel de classe e suficiente...";
			mes "Bom. Agora vou te dar a qualificacao para se tornar um Acolito.";
			next;
			mes "[Father Mareusis]";
			mes "Sempre lembre-se de ser grato a Deus, que cuida de nos o tempo todo. No caos e em tempos de dificuldade, enfrente suas provacoes com fe inabalavel.";
			callfunc "Job_Change",Job_Acolyte;
			getitem 1545,1;	// N_Mace
			next;
			mes "[Father Mareusis]";
			mes "Por ultimo, quero sinceramente parabeniza-lo por perseverar em sua prova de penitencia.";
			close;
		}
	case 2:
		mes "[Father Mareusis]";
		mes "Voce deseja se tornar um Acolito?";
		mes "Entao, voce deve cumprir os seguintes requisitos pensando que sao as praticas dadas por Deus.";
		next;
		mes "[Father Mareusis]";
		mes "Primeiro, voce deve alcancar pelo menos o Nivel de Classe de Aprendiz 9 e aprender todas as Habilidades Basicas.";
		mes "Isso e o mais basico a se fazer, entao voce precisa encarar como forma de treinar a si mesmo.";
		next;
		mes "[Father Mareusis]";
		mes "Quando achar que cumpriu esse requisito, volte a mim novamente. Entao voce tera uma classe sagrada na qual podera espalhar a vontade de Deus.";
		close;
	case 3:
		close;
	}
}
