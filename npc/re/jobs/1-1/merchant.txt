		if (Class == Job_Novice_High && (ADVJOB == Job_Whitesmith || ADVJOB == Job_Creator)) {
			mes "[Chief Mahnsoo]";
			mes "Quanto tempo!";
			mes "Ei, voce nao desistiu";
			mes "do seu negocio, desistiu?";
			mes "O que aconteceu?";
			next;
			mes "[Chief Mahnsoo]";
			mes "Uau...";
			mes "Voce realmente foi a Valhalla?! Caramba, voce progrediu muito...";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Chief Mahnsoo]";
				mes "Hmmm...";
				mes "Parece que voce nao esta pronto para se tornar Mercador novamente. Va terminar de aprender as Habilidades Basicas de Aprendiz primeiro.";
				next;
				mes "[Chief Mahnsoo]";
				mes "Nao se preocupe, sempre teremos uma vaga de Mercador para voce. Volte quando estiver pronto, ta?";
				close;
			}
			mes "[Chief Mahnsoo]";
			mes "Acho que e o destino nos encontrarmos assim mais uma vez. Certo. Mais uma vez, deixe-me transforma-lo em Mercador!";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Merchant_High;
			skill "MC_CARTREVOLUTION",1,SKILL_PERM;
			skill "MC_CHANGECART",1,SKILL_PERM;
			skill "MC_LOUD",1,SKILL_PERM;
			mes "[Chief Mahnsoo]";
			mes "Ah~ Que nostalgico. Igualzinho aos velhos tempos! Certo, de o seu melhor!";
			close;
		}else{
			mes "[Chief Mahnsoo]";
			mes "^333333*Suspiro*^000000";
			mes "Estou tao entediado...";
			mes "Quando receberei noticias da minha querida Blossom?";
			close;
		}
	}
	mes "[Guildsman Mahnsoo]";
	mes "Ei, por que voce esta aqui?";
	next;
	switch(select("Quero ser um mercador.:Quero saber mais sobre mercadores.:Perguntar os requisitos para ser mercador.:Nada.")) {
	case 1:
		mes "[Guildsman Mahnsoo]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Merchant) {
				mes ".....? Desculpe? O que voce esta dizendo?";
				mes "Voce ja e um mercador. Meu Deus..";
				mes "Ha?! ...Eu deveria rir agora?!";
			}else{
				mes "Ahh? Voce esta tentando ter os dois lados?";
				mes "Que tal simplesmente se dedicar a sua classe original?";
				next;
				mes "[Guildsman Mahnsoo]";
				mes "Nos temos etica nos negocios, sabe.";
			}
			close;
		}else{
			mes "Quer ser um mercador? Hmm...";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Guildsman Mahnsoo]";
				mes "Mas se voce quer ser mercador, seu nivel de habilidade basica deve chegar ao Nivel 9 ou voce deve gastar todos os seus pontos de habilidade.";
				next;
				mes "[Guildsman Mahnsoo]";
				mes "Voce nao acha que precisamos aprender algumas habilidades basicas mesmo que so lidemos com dinheiro?";
				close;
			}
			mes "[Guildsman Mahnsoo]";
			mes "Bom, acho que voce esta totalmente pronto vendo que seu nivel de habilidade basica esta cumprido.";
			mes "Agora eu permito que voce seja um mercador.";
			next;
			mes "[Guildsman Mahnsoo]";
			mes "Parabens por se tornar um mercador!";
			mes "Parabens novamente por ser membro da guilda dos mercadores e um dos nossos colegas. Espero sua participacao ativa de agora em diante!";
			callfunc "Job_Change",Job_Merchant;
			getitem 1381,1;	// N_Battle_Axe
			next;
			mes "[Guildsman Mahnsoo]";
			mes "Com certeza, precisamos de jovens que tenham paixao para alcancar nosso grande objetivo de garantir 20% do volume monetario mundial. Entendeu? Hein?";
			mes "Bem, so estou dizendo... no final significa vamos ganhar muito dinheiro. Voces sabem disso~";
			close;
		}
	case 2:
		mes "[Guildsman Mahnsoo]";
		mes "Mercador? Simplificando, a pessoa que vende mercadorias e ganha dinheiro e um mercador.";
		mes "Nao e bom em lutar e nao tem habilidades especiais de ataque/recuperacao... mas um mercador pode comprar coisas a preco baixo e depois vende-las e ganhar dinheiro.";
		next;
		mes "[Guildsman Mahnsoo]";
		mes "Bem, um mercador tem uma habilidade suprema chamada Mammonite que atinge o inimigo com seu dinheiro... Podemos equipar tudo exceto Arcos, Varinhas e Espadas de Duas Maos. Mas sempre podemos vender e comprar essas.";
		mes "Sim... nos mercadores sempre temos dinheiro em mente, entendeu?";
		close;
	case 3:
		mes "[Guildsman Mahnsoo]";
		mes "Para se tornar mercador, embora apenas vender e receber dinheiro seja nosso trabalho, voce deve alcancar pelo menos o nivel 9 de habilidade basica.";
		next;
		mes "[Guildsman Mahnsoo]";
		mes "Bem, nos costavamos cobrar uma taxa de inscricao antes. Nao era dificil juntar aquele dinheiro?";
		mes "Tinha gente demais que nao conseguia juntar aquele dinheiro e ficava chorando.";
		next;
		mes "[Guildsman Mahnsoo]";
		mes "Entao recentemente, decidimos permitir que qualquer Aprendiz que queira ser mercador se torne um.";
		mes "Porque estar vivo ou nao depois depende da habilidade de cada um.";
		close;
	case 4:
		close;
	}
}
