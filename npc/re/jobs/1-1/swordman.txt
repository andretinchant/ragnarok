izlude_in,74,172,4	script	Swordman#swd	119,{
	if (Upper == 1) {
		if (Class == Job_Novice_High && (ADVJOB == Job_Lord_Knight || ADVJOB == Job_Paladin)) {
			mes "[Swordman]";
			mes "Nao...";
			mes "Pode ser...";
			mes "Voce renasceu, nao foi?";
			next;
			mes "[Swordman]";
			mes "Vejo que voce esta trilhando novamente o caminho do Espadachim! Uma vez que voce se acostuma a empunhar uma espada, nao tem como voltar atras!!";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Swordman]";
				mes "Hmm? Ah, voce precisa dominar as Habilidades Basicas antes de estar pronto para se tornar um Espadachim.";
				next;
				mes "[Swordman]";
				mes "Volte quando tiver terminado de aprender as Habilidades Basicas de Aprendiz.";
				close;
			}
			mes "[Swordman]";
			mes "Excelente! Deixe-me promove-lo a Espadachim agora mesmo!";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Swordman_High;
			skill "SM_MOVINGRECOVERY",1,SKILL_PERM;
			skill "SM_FATALBLOW",1,SKILL_PERM;
			skill "SM_AUTOBERSERK",1,SKILL_PERM;
			mes "[Swordman]";
			mes "Hmm... Voce parece um Espadachim bem experiente. Ainda assim, tenho certeza de que voce precisa treinar para melhorar suas habilidades e ganhar forca!";
			close;
		}else{
			mes "[Swordman]";
			mes "Hm...?";
			mes "Voce e um guerreiro";
			mes "renascido, nao e?";
			next;
			mes "[Swordman]";
			mes "Hmmm...";
			mes "Parece que ser";
			mes "um Espadachim nao faz parte";
			mes "do seu destino. Sinto muito,";
			mes "mas parece que nao ha nada";
			mes "que eu possa fazer por voce.";
			close;
		}
	}
	mes "[Swordman Guildsman]";
	mes "Esta e a Guilda dos Espadachins.";
	mes "Por que voce esta aqui?";
	next;
	switch(select("Me fale sobre ser Espadachim.:Quero ser um Espadachim.:Nada.")) {
	case 1:
		mes "[Swordman Guildsman]";
		mes "Entao voce quer saber mais sobre o poderoso Espadachim? Certo!";
		next;
		mes "[Swordman Guildsman]";
		mes "A caracteristica mais marcante do Espadachim e que ele pode mostrar suas verdadeiras habilidades em combate corpo a corpo.";
		mes "Ha tres razoes!";
		next;
		mes "[Swordman Guildsman]";
		mes "Primeiro, o Espadachim tem HP mais alto que outras classes.";
		mes "Segundo, exceto Arcos e Varinhas, o Espadachim pode usar todas as outras armas para lutar em sua capacidade ideal.";
		mes "E terceiro, a maioria das habilidades do Espadachim causa ataques fisicos poderosos.";
		next;
		mes "[Swordman Guildsman]";
		mes "Embora eu tenha dado uma explicacao simples, acredito que voce entendeu o significado principal do que e ser um Espadachim.";
		mes "Na minha opiniao, Espadachim e a melhor classe de todas!";
		close;
	case 2:
		mes "[Swordman Guildsman]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Swordman) {
				mes "Voce ja e um excelente Espadachim, nao e?";
				mes "Apenas se dedique a ser um grande Espadachim.";
			}else{
				mes "Voce ja tem outra classe, nao e?";
				mes "Voce foi longe demais com essa brincadeira.";
			}
			close;
		}else{
			if (!callfunc("F_CanChangeJob")) {
				mes "Sinto muito em te dizer isso, mas para ser Espadachim, voce deve alcancar pelo menos ^4A4AFFNivel de Classe 10^000000.";
				mes "e ^4A4AFFNivel de Habilidade Basica 9^000000.";
				next;
				mes "[Swordman Guildsman]";
				mes "Quer ser Espadachim sem ter o requisito minimo?";
				mes "Voce acha que ser Espadachim e tao facil assim?";
				close;
			}
			mes "Hmm, tanto seu Nivel de Classe quanto seu Nivel de Habilidade Basica estao corretos.";
			mes "Bom. Voce quer se tornar Espadachim agora mesmo?";
			next;
			if(select("Sim, eu quero.:Vou pensar de novo.") == 2) {
				mes "[Swordman Guildsman]";
				mes "Sim. Uma decisao prudente e necessaria para escolher uma classe.";
				mes "Mas eu sinto muito... que voce reconsidere depois de superar todas as dificuldades....";
				close;
			}
			mes "[Swordman Guildsman]";
			mes "Parabens! De agora em diante, voce vai viver a vida de um Espadachim!";
			mes "Vamos fazer isso agora mesmo!";
			next;
			mes "[Swordman Guildsman]";
			mes "Parabens novamente por ser um Espadachim e espero que voce participe de muitas atividades para o reavivamento da nossa guilda.";
			callfunc "Job_Change",Job_Swordman;
			getitem 13415,1; // N_Falchion
			close;
		}
	case 3:
		close;
	}
}
