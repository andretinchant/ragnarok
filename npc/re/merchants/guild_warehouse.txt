-	script	GuildWarehouse	-1,{
	mes "[Warehouse Manager]";
	mes "Como vai? Nos somos especializados em armazens de guilda. Essa e a nossa ^0000cdhistoria^000000:";
	mes "Por que os membros da guilda nao podem compartilhar um armazem? Comecamos com essa simples pergunta.";
	next;
	mes "[Warehouse Manager]";
	mes "Voce pode abrir o armazem por 1000 Zeny ou consultar o historico de uso da guilda.";
	mes "Como posso te ajudar?";
	next;
	if( select( "Abrir armazem da guilda:Ver historico de uso do armazem" ) == 1 ){
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[Warehouse Manager]";
			mes "O armazem da guilda so esta disponivel para membros de guilda.";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[Warehouse Manager]";
			mes "Parece que ainda nao e possivel para sua guilda usar o armazem da guilda ou voce nao tem permissoes de acesso ao armazem.";
			mes "Por favor, volte apos verificar a habilidade da guilda e a permissao de acesso ao armazem.";
			close;
		}

		if( Zeny < 1000 ){
			mes "[Warehouse Manager]";
			mes "Receio que voce nao tenha dinheiro suficiente para pagar a taxa.";
			mes "A taxa e de 1000 Zeny.";
			close;
		}

		mes "[Warehouse Manager]";
		mes "Vou abrir o armazem da guilda pra voce entao. Tenha um momento memoravel!";
		close2;

		if( Zeny < 1000 ){
			// Cheat prevention
			end;
		}

		if( guildopenstorage() == GSTORAGE_OPEN ){
			Zeny -= 1000;
			end;
		}else{
			mes "[Warehouse Manager]";
			mes "Receio que outra pessoa esteja usando o armazem agora.";
			mes "Por favor, tente novamente depois de um tempo.";
			close;
		}
	}else{
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[Warehouse Manager]";
			mes "O armazem da guilda so esta disponivel para membros de guilda.";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[Warehouse Manager]";
			mes "Parece que ainda nao e possivel para sua guilda usar o armazem da guilda ou voce nao tem permissoes de acesso ao armazem.";
			mes "Por favor, volte apos verificar a habilidade da guilda e a permissao de acesso ao armazem.";
			close;
		}

		mes "[Warehouse Manager]";
		mes "Vou te mostrar o historico de uso do armazem. O historico de uso sera mantido por ate 3 meses.";
		mes "Tenha um momento memoravel!";
		close2;
		guildopenstorage_log();
		end;
	}
}
