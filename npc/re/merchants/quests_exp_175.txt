einbech,149,257,4	script	Site manager Elmen#extdq0	4_TOWER_16,{
	if (checkweight(25814,100) == 0 || checkweight(1201,1) == 0 || (MaxWeight - Weight) < 10000) {
		mes "- Voce nao pode prosseguir porque tem itens demais. -";
		close;
	}
	mes "[Elmen]";
	mes "Ola. Aventureiro.";
	mes "Se voce esta aqui para comprar equipamento de mineracao de Dynite ou refinar minerio em Dynite, voce veio ao lugar certo.";
	next;
	switch( select( "Refinamento de Dynite", "Loja de Dynite", "Troca de Equipamento de Dynite" ) ) {
	case 1:
		npctalk "Elmen: Voce pode refinar ate 100 Dynite por vez. A maquina e pequena.", "", bc_self;
		mes "[Elmen]";
		mes "" + mesitemlink(25814) + " Minerio Refinado.";
		mes "" + mesitemlink(25809) + " ^777777(atualmente " + countitem(25809) + ")^000000";
		mes "" + mesitemlink(25810) + " ^777777(atualmente " + countitem(25810) + ")^000000";
		mes "" + mesitemlink(25811) + " ^777777(atualmente " + countitem(25811) + ")^000000";
		mes "" + mesitemlink(25812) + " ^777777(atualmente " + countitem(25812) + ")^000000";
		mes "" + mesitemlink(25813) + " ^777777(atualmente " + countitem(25813) + ")^000000";
		mes "Voce precisa de um de cada um dos cinco tipos de minerio.";
		next;
		input .@val;
		if (.@val > 100) {
			mes "[Elmen]";
			mes "Sinto muito. Aventureiro.";
			mes "O Refinamento de Dynite so pode refinar ate 100 por vez. A maquina e pequena, entao nao tem jeito. Por favor, entenda.";
			close;
		}
		if (.@val == 0) {
			mes "[Elmen]";
			mes "Cancelar?";
			mes "Se precisar de refinamento de Dynite, me procure a qualquer momento. Ninguem mais consegue fazer isso alem de mim.";
			close;
		}
		if (countitem(25809) < .@val || countitem(25810) < .@val || countitem(25811) < .@val || countitem(25812) < .@val || countitem(25813) < .@val) {
			mes "[Elmen]";
			mes "Ei! Isso dificulta as coisas pra nos dois, voce nao esta sem os materiais necessarios para o refinamento? Os materiais!";
			close;
		}
		specialeffect EF_REPAIRWEAPON;
		progressbar_npc "FFFF00",1;
		delitem 25809,.@val;	// Ein_REDIUM
		delitem 25810,.@val;	// Ein_RINDIUM
		delitem 25811,.@val;	// Ein_ODIUM
		delitem 25812,.@val;	// Ein_PURDIUM
		delitem 25813,.@val;	// Ein_WHIDIUM
		getitem 25814,.@val;	// Ein_DYNITE
		mes "[Elmen]";
		mes "" + .@val + " Dynite foram refinados. Esta perfeito hoje!";
		close;
	case 2:
		mes "[Elmen]";
		mes "Ha varios acessorios e permissoes de modificacao. Olhe bem o que voce quer antes de comprar.";
		close2;
		callshop "barter_ein_dun03";
		end;
	case 3:
		mes "[Elmen]";
		mes "Eu posso trocar o equipamento pelo Dynite, mas ja esta em uma caixa bem lacrada.";
		mes "Me de 50 ^4d4dffDynites^000000 e eu abrirei uma ^4d4dffcaixa de arma^000000 pra voce.";
		next;
		mes "[Elmen]";
		mes "Eu nao sei o que vai sair, mas a qualidade e boa. Voce gostaria de abrir?";
		next;
		if (select( "Fechar os olhos e trocar.", "Nao trocar." ) == 2) {
			mes "[Elmen]";
			mes "Kee, que desperdicio, era um bom negocio.";
			close;
		}
		if (countitem(25814) < 50) {
			mes "[Elmen]";
			mes "Voce nao tem Dynite suficiente, amigo.";
			mes "Eu nao sei o que vai sair, mas e dificil conseguir esse tipo de equipamento por esse preco em outro lugar.";
			close;
		}
		delitem 25814,50;
		// unknown text
		getgroupitem(IG_EIN_1HITEMBOX);
		end;
	}
}
