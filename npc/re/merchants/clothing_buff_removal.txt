//===== rAthena Script =======================================
//= Clothing Effect Removal Service
//===== Description: =========================================
//= Service to remove clothing effects like Xmas, Wedding, ...
//===== Changelog: ===========================================
//= 1.0 Initial version. [Balferian, Everade, Lemongrass]
//============================================================

prontera,80,106,5	script	Clothing Effect Removal Service	4_EP16_COOK2,{
	mes "[Derek Shot]";
	mes "Voce nao consegue participar de batalhas por causa de buffs de roupa? Deve ser irritante.";
	next;
	mes "[Derek Shot]";
	mes "Embora fique bem em voce, tambem pode ser muito irritante. Eu posso me livrar disso com um unico gesto de mao!! Vou arrumar tudo direitinho. Se voce estiver pronto, escolha sua opcao!";
	next;
	switch( select( "Manter assim", "Remover imediatamente" ) ){
		case 1:
			mes "[Derek Shot]";
			mes "Volte pra mim se mudar de ideia.";
			close;
		case 2:
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			mes "[Derek Shot]";
			mes "A remocao foi concluida. Certifique-se de que pode se mover livremente agora.";
			atcommand "@changedress";
			close;
	}
}
