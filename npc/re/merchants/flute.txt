-	script	Falcon Flute Trader::fflute	-1,{
	mes "[Falcon Flute Trader]";
	if (BaseJob == Job_Hunter) {
		mes "Voce precisa de um Falcao?";
		mes "Com uma Flauta magica, voce pode chamar seu Falcao de qualquer lugar!";
		next;
		mes "[Falcon Flute Trader]";
		mes "E maravilhoso, nao e? Ha ha ha!";
		next;
		mes "[Falcon Flute Trader]";
		mes "Por que estou vendendo essas Flautas de Falcao?";
		mes "E porque eu nao faco ideia de como elas funcionam.";
		mes "Ha ha ha!";
		next;
		mes "[Falcon Flute Trader]";
		mes "Claro, voce vai precisar da Habilidade ^ff0000Falcon Mastery^000000.";
		mes "Eu nao tenho a Habilidade, entao nao sei como lidar com Falcoes.";
		mes "Do jeito que esta agora, eu talvez nunca tenha um Falcao.";
		next;
		mes "[Falcon Flute Trader]";
		mes "Mas eu nao posso te dar uma de graca!";
		mes "Eu tenho que ganhar dinheiro ja que tambem preciso comer!";
		next;
		mes "[Falcon Flute Trader]";
		mes "Se voce precisar de uma Flauta de Falcao eu vendo por 12.500 zeny. Ou isso e muito caro?";
		next;
		if (eaclass()&EAJL_THIRD) {
			mes "[Falcon Flute Trader]";
			mes "Para Rangers, porem, eu tenho um preco especial de 10.000 zeny!";
			set .@price,10000;
			next;
		} else
			set .@price,12500;
		if(select("Nao comprar:Comprar uma Flauta de Falcao") == 1) {
			mes "[Falcon Flute Trader]";
			mes "Que pena. Se precisar de uma, volte!";
			close;
		}
		if (Zeny >= .@price) {
			if (countitem(12848) >= 1) {
				mes "[Falcon Flute Trader]";
				mes "Pelo que vejo voce ja tem uma Flauta de Falcao!";
				mes "Nao se preocupe, meu jovem amigo, uma vez que voce comprou a Flauta de Falcao nao precisara comprar de novo";
				mes "ja que a Flauta de Falcao nao desaparece apos o uso, nao e otimo?";
				close;
			}
			mes "[Falcon Flute Trader]";
			mes "Aqui, pegue esta flauta.";
			mes "Com ela voce pode assobiar para seu Falcao de qualquer lugar.";
			set Zeny, Zeny - .@price;
			getitem 12848,1; //Falcon_Flute
			next;
			mes "[Falcon Flute Trader]";
			mes "So um lembrete.";
			mes "Voce precisa da Habilidade ^ff0000Falcon Mastery^000000 para poder usar!";
			close;
		}
		mes "[Falcon Flute Trader]";
		mes "Sinto muito mas voce nao tem dinheiro suficiente para comprar uma Flauta de Falcao, por favor volte quando tiver mais dinheiro!";
		close;
	}
	if (BaseJob == Job_Archer) {
		mes "Ola, jovem!";
		mes "No momento nao posso te ajudar,";
		mes "mas por que voce nao volta pra mim quando se tornar um grande Cacador e aprender a lidar com Falcoes direito?";
		close;
	}
	mes "Nao e um dia lindo hoje?";
	close;
}
