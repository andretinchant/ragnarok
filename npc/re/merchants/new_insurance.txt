prontera,82,108,5	script	Heart Merchant#life01	4_M_NFDEADMAN2,5,5,{
	.@old_item = 12209;
	.@new_item = 6413;
	.@old_item_name$ = getitemname(.@old_item);
	.@new_item_name$ = getitemname(.@new_item);

	if (!checkweight(1201,1) || (MaxWeight - Weight) < 500) {
		mes "Voce nao pode prosseguir com a conversa porque tem muitos itens.";
		mes "Por favor, organize seus itens e tente novamente.";
		close;
	}
	mes "[Heart Merchant]";
	mes "Ola! Vejo que voce e um aventureiro que dedica o coracao a viagens, usa barcos para navegar pelo continente, e aproveita a emocao do perigo!";
	next;
	if(!countitem(.@old_item)){
		mes "[Heart Merchant]";
		mes "Voce tem algum " + mesitemlink( .@old_item ) + " no seu armazem?";
		next;
		mes "[Heart Merchant]";
		mes "Se voce me trouxer um " + .@old_item_name$ + ", eu troco por " + mesitemlink( .@new_item ) + ". O que acha? Voce tem ideia do que esse item e?";
	} else {
		mes "[Heart Merchant]";
		mes "Voce ja pensou em trocar seu " + mesitemlink( .@old_item ) + " por algo melhor? Por exemplo... " + mesitemlink( .@new_item ) + ". O que acha?";
	}
	next;
	switch(select("O que e isso?",(countitem(.@old_item)?"Troque, por favor!":""),"Nao estou interessado")) {
	case 1:
		mes "[Heart Merchant]";
		mes mesitemlink( .@old_item ) + " e um item que impede voce de perder experiencia ao morrer uma vez dentro de 30 minutos de duracao apos o uso.";
		next;
		mes "[Heart Merchant]";
		mes "Mas muitas pessoas ficam tristes com o limite de tempo, nao e? Entao eu pesquisei e preparei o " + mesitemlink( .@new_item ) + ".";
		next;
		mes "[Heart Merchant]";
		mes mesitemlink( .@new_item ) + " e um item que vai impedir a perda de experiencia ao morrer uma vez apenas por te-lo ^EE0000independente da duracao^000000. Claro, apos a morte, o "+.@new_item_name$+" e consumido.";
		next;
		select("Por que voce esta trocando?");
		mes "[Heart Merchant]";
		mes "Voce esta curioso sobre isso? Eu sou um mercador. Comprar e vender coisas e meu destino, seja relacionado a vida ou a morte. Voce acha que eu vou perder essa oportunidade de ganhar dinheiro? Eu ganho dinheiro, e os vivos ganham um alivio da dor da morte. Nao e um bom negocio?";
		next;
		mes "[Heart Merchant]";
		mes "A taxa de troca e 5 " + mesitemlink( .@old_item ) + " por 1 " + mesitemlink( .@new_item ) + ". Voce tem alguma pergunta?";
		next;
		mes "[Heart Merchant]";
		mes "Hm~ O que voce acha? Nao e uma troca justa? Voce quer ficar sempre preocupado com o tempo? E um item indispensavel para aventureiros que gostam de aventuras emocionantes.";
		close;

	case 2:
		mes "[Heart Merchant]";
		mes "Isso, isso. Boa decisao. E melhor trocar por um melhor, nao e?";
		next;
		mes "[Heart Merchant]";
		mes "Eu posso trocar seus ^0000cd5 "+.@old_item_name$+" por 1 "+.@new_item_name$+".";
		next;
		mes "[Heart Merchant]";
		mes "Porem, ^EE0000[Nao vendavel] " + .@old_item_name$ + " nao e elegivel para troca^000000, entao e inutil trazer aqui e pedir para trocar.";
		next;
		mes "[Heart Merchant]";
		mes "Vamos trocar agora?";
		next;
		if (select("Troque, por favor!:Nao quero trocar.") == 2)
			break;
		mes "[Heart Merchant]";
		mes "Quantos voce quer trocar? ^0000cdVoce pode trocar ate 10 "+.@new_item_name$+" por vez^000000.";
		next;
		input .@amount;
		if (.@amount == 0) {
			mes "[Heart Merchant]";
			mes "Hm? Voce nao esta interessado? Esse e um otimo produto, e e muito bom pra alguem como voce. Mas se voce nao quer proteger sua vida, eu tambem nao vou te forcar.";
			close;
		}
		if (.@amount > 10) {
			mes "[Heart Merchant]";
			mes "Como eu disse, ^0000cdvoce so pode trocar ate 10^000000 por vez. Verifique a quantidade novamente.";
			close;
		}
		if (countitem(.@old_item) < (.@amount*5)) {
			mes "[Heart Merchant]";
			mes "Hm, com licenca mas... voce nao tem " +.@old_item_name$+ " suficientes, nao? Sao 5 " + mesitemlink( .@old_item ) + " por 1 " + mesitemlink( .@new_item ) + ". Por favor, verifique a quantidade.";
			close;
		}
		mes "[Heart Merchant]";
		mes "Pronto, com isso, eu troquei seu ^0000cd" + .@old_item_name$ + "^000000 pelo ^EE0000" + .@new_item_name$ + "^000000!! Foi um bom negocio, nao foi? Da proxima vez que conseguir um "+.@old_item_name$+", venha me ver de novo!";
		delitem .@old_item,.@amount*5;
		getitem .@new_item,.@amount;
		close;
	case 3:
		break;
	}
	mes "[Heart Merchant]";
	mes "Esse e um otimo negocio! Aventureiro, e um item indispensavel para pessoas como voce que se aventuram em lugares perigosos!";
	next;
	mes "[Heart Merchant]";
	mes "Se voce ficar interessado, volte com certeza para a troca. Ta bom?";
	close;

OnTouch:
	if(countitem(12209))
		npctalk "Voce tem algo bom. Gostaria de trocar seu " + getitemname(12209) + " por algo melhor?", "", bc_self;
	end;
}
