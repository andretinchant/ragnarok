//= arg(1): How many of each 'getarg(0)' arrow per quiver.
//= arg(2): The cost of making a 'getarg(0)' quiver.

// Mora :: ep14_1_tre
//============================================================
mora,106,117,3	script	Quiver Maker#mora	4_M_RAFLE_GR,{
	if (checkweight(1201,1) == 0) {
		mes "[Quiver Maker]";
		mes "Voce tem coisas demais com voce.";
		mes "Libere espaco no seu inventario e volte. Vou te contar algo interessante.";
		close;
	}
	if (MaxWeight - Weight < 2000) {
		mes "[Quiver Maker]";
		mes "Voce parece esgotado com tanta coisa.";
		mes "Libere espaco no seu inventario e volte. Vou te contar algo interessante.";
		close;
	}
	mes "[Quiver Maker]";
	mes "Os moradores de Mora perguntam pra que servem aljavas. Eles simplesmente nao sabem como o mundo funciona.";
	mes "Nao e de admirar que nao saibam nada sobre aljavas - passando geleia nas folhas o dia todo.";
	next;
	if (getbattleflag("feature.barter")) {
		.@menu$ = "Eu quero flechas";
	}
	switch( select("Por favor, faca uma aljava pra mim.", .@menu$, "O que e uma aljava?") ) {
	case 1:
		if (getbattleflag("feature.barter")) {
			mes "[Quiver Maker]";
			mes "Finalmente, alguem apareceu que conhece o verdadeiro valor de uma aljava. Eu farei para aqueles que conhecem seu valor.";
			close2;
			callshop "Quiver_Archer", 1;
			end;
		}
		mes "[Quiver Maker]";
		mes "Finalmente alguem que aprecia uma aljava!";
		mes "Eu posso fazer Aljavas Elficas e Aljavas de Caca.";
		mes "Qual voce precisa?";
		next;
		switch(select("Uma Aljava Elfica:Uma Aljava de Caca:Nao preciso de aljava.")) {
		case 1: callsub S_BuyQuiver,1773,500,500,12575; //Arrow_Of_Elf_Cntr
		case 2: callsub S_BuyQuiver,1774,500,500,12576; //Hunting_Arrow_Cntr
		case 3:
			mes "[Quiver Maker]";
			mes "Voce pode comprar flechas do mercador de ferramentas ao meu lado.";
			close;
		}
		end;
	case 2:
		if (getbattleflag("feature.barter")) {
			mes "[Quiver Maker]";
			mes "Eu nao vendo flechas comuns. Essas sao flechas artesanais que so podem ser compradas aqui. Voce realmente reconheceria seu valor, certo?";
			close2;
			callshop "Catch_Arrows", 1;
		}
		end;
	case 3:
		mes "[Quiver Maker]";
		mes "Uma flecha pode ser fina e leve, mas carregar centenas e milhares de flechas e como carregar um tronco inteiro.";
		next;
		mes "[Quiver Maker]";
		mes "Mas quando voce tem aljavas, pode colocar flechas nelas e economizar peso e espaco.";
		mes "Se estiver interessado em uma, eu costuro uma pra voce.";
		close;
	}
	end;

S_BuyQuiver:
	.@item_req = getarg(0);
	.@req_amount = getarg(1);
	if (countitem(getarg(0)) < .@req_amount) {
		mes "[Quiver Maker]";
		mes "Traga mais de "+ .@req_amount +" "+ getitemname(.@item_req) +" e eu faco uma aljava pra voce.";
		close;
	}
	mes "[Quiver Maker]";
	mes "Ah, vejo que voce tem "+ getitemname(.@item_req) +" na mao!";
	mes "Voce esta interessado em usar uma aljava? E realmente conveniente!";
	mes "Se estiver interessado, eu posso trocar "+ .@req_amount +" dessas flechas por uma dessas aljavas por ^FF3131"+getarg(2)+" zeny^000000.";
	next;
	switch(select("Trocar todas as flechas que voce tem:Pegar apenas uma aljava:Nao trocar")) {
	case 1:
		set .@arrows, countitem(.@item_req);
		set .@quiver, .@arrows / .@req_amount;
		set .@arrows_used, .@quiver * .@req_amount;
		set .@arrow_zeny01, .@quiver * getarg(2);
		mes "Numero de flechas que voce tem : ^3131FF"+.@arrows+"^000000";
		mes "Numero de aljavas disponiveis : ^3131FF"+.@quiver+"^000000";
		mes "Zeny necessario para troca : ^3131FF"+.@arrow_zeny01+" zeny^000000";
		next;
		mes "Trocar?";
		next;
		if(select("Trocar:Nao trocar") == 2) {
			mes "[Quiver Maker]";
			mes "Ei, voce nao duvida das minhas habilidades, ne?";
			close;
		}
		break;
	case 2:
		set .@quiver, 1;
		set .@arrows_used, .@req_amount;
		set .@arrow_zeny01, getarg(2);
		set .@zeny_mes,1;
		break;
	case 3:
		mes "[Quiver Maker]";
		mes "Ei, voce nao duvida das minhas habilidades, ne?";
		close;
	}
	if (Zeny < .@arrow_zeny01) {
		mes "[Quiver Maker]";
		if (.@zeny_mes == 1)
			mes "Eu disse que aceitaria moedas humanas so por voce, e ainda nao quer gastar o dinheiro?";
		else
			mes "Voce realmente nao espera colocar as maos numa obra-prima de graca, ne?";
		close;
	}
	mes "[Quiver Maker]";
	mes "Ei, aqui esta.";
	mes "Ha ^3131FFalgo que voce precisa saber^000000 - tente lembrar.";
	mes "^FF0000Voce nao pode usar aljavas quando sua carga esta acima de 70%.^000000";
	mes "E melhor guardar isso na cabeca, ou voce pode ter problemas depois.";
	set Zeny, Zeny-.@arrow_zeny01;
	delitem .@item_req,.@arrows_used;
	getitem getarg(3),.@quiver;
	close;
}
