moc_para01,22,16,5	script	Catalog Magician#catal01	704,{
	if (!checkweight(1301,3)) {
		mes "- Voce tem itens demais. Por favor, libere espaco. -";
		close;
	}
	set .@ticket_cost, 200;
	mes "[Catalog Magician]";
	mes "Olha... a Academia de Magia";
	mes "em Geffen agora esta vendendo";
	mes "diretamente o novissimo";
	mes "^007777Catalogo Universal Prata^000000!";
	next;
	mes "Ele esta olhando para as palavras";
	mes "escritas na mao dele.";
	mes "Ele parece muito ocupado";
	mes "para olhar pra voce.";
	next;
	mes "[Catalog Magician]";
	mes "Um catalogo novinho em folha";
	mes "acabou de sair!";
	mes "Voce pode verificar";
	mes "quais itens pode trocar";
	mes "e quais itens estao a venda";
	mes "nos vendedores.";
	next;
	mes "[Catalog Magician]";
	mes "O ^007777Catalogo Universal Prata^000000";
	mes "custa apenas 200z!";
	mes "Voce pode comprar ate 50 unidades de uma vez!";
	next;
	switch(select("Comprar um ^007777Catalogo Universal Prata^000000:Nao preciso!:Voce nao parece um vendedor!")) {
	case 1:
		mes "[Catalog Magician]";
		mes "Voce... quer todas as 50 unidades?";
		mes "Ou quantas voce precisa?";
		next;
		while (1) {
			input .@input;
			mes "[Catalog Magician]";
			if (.@input == 0) {
				mes "A negociacao foi cancelada!";
				mes "Eu nao sei o que fazer agora...";
				mes "O que eu deveria fazer?";
				emotion ET_PROFUSELY_SWEAT, playerattached();
				close;
			} else if (.@input < 1 || .@input > 50) {
				mes "Devem ser menos de 50 unidades.";
				next;
			} else
				break;
		}
		set .@sell, .@ticket_cost * .@input;
		mes "O numero total de catalogo(s) que voce esta tentando comprar e " + .@input + " unidades.";
		mes "Custa " + .@sell + " z.";
		if (Zeny < .@sell) {
			mes "Parece que voce nao tem dinheiro suficiente.";
			mes "Voce pode vender seus equipamentos";
			mes "para ganhar mais dinheiro... se quiser.";
			close;
		}
		mes "Ok, aqui esta.";
		set Zeny, Zeny - .@sell;
		getitem 12580,.@input; //Vending_Search_Scroll
		close;
	case 2:
		mes "[Catalog Magician]";
		mes "Argh... voce realmente precisa desse";
		mes "item... Quer que eu";
		mes "explique de novo? Eu nao sei";
		mes "o que fazer agora...";
		mes "O que eu deveria fazer?";
		emotion ET_PROFUSELY_SWEAT, playerattached();
		close;
	case 3:
		mes "[Catalog Magician]";
		mes "Meu verdadeiro trabalho e ser mago.";
		mes "Eu ia pedir pra guilda de mercadores";
		mes "vender eles, mas eles estavam";
		mes "tentando usar como um adendo";
		mes "nas vendas de vassouras. Por isso";
		mes "eu estou vendendo por conta propria.";
		next;
		mes "[Catalog Magician]";
		mes "Ha muitos aventureiros";
		mes "aqui que vem e vao.";
		mes "Tenho certeza de que posso vender pra todos eles em breve.";
		mes "Voce nao acha?";
		mes "Entao... vamos comecar!! Compre mais deles, por favor! POR FAVOR...!";
		close;
	}
}
