grademk,34,184,4	script	Sratos#sratos	4_JP_GARM_H,{
	mes "[Sratos]";
	mes "Ola, querido cliente que entrou, vamos torcer por um milagre hoje.";
	mes "Como posso te ajudar?";
	next;
	switch( select( "Aprimorar o grau do equipamento.", "Trocar itens Etel", "O que e Aprimoramento de Grau?" ) ){
		case 1:
			mes "[Sratos]";
			mes "Nosso cliente. Voce quer aprimorar o grau do equipamento.";
			mes "Nao e facil mover o poder magico das joias.";
			next;
			mes "[Sratos]";
			mes "O poder magico pode explodir na hora.";
			mes "E entao a arma que vai herdar o poder magico tambem explode!";
			next;
			switch( select( "Vou fazer mesmo assim!", "Vou pensar sobre isso." ) ){
				case 1:
					mes "[Sratos]";
					mes "Desejo boa sorte aos nossos corajosos clientes!";
					close2;
					enchantgradeui();
					end;
				case 2:
					mes "[Sratos]";
					mes "Quando voce tiver coragem de desafiar, por favor volte.";
					close;
			}
		case 2:
			mes "[Sratos]";
			mes "Essas sao joias usadas para aprimorar graus.";
			mes "Etel Dust, as joias que serao a base, e se voce me der uma pequena taxa eu troco por joias Etel.";
			close2;
			callshop "EnchantGradeExchange";
			end;

		case 3:
			mes "[Sratos]";
			mes "Ocasionalmente, joias instaveis com muito poder magico puro sao encontradas.";
			mes "Nos chamamos essas de gemas etericas, certo?";
			next;
			mes "[Sratos]";
			mes "Voce pode transferir o poder magico da gema eterica para outras armas, mas claro que ha algum risco.";
			mes "E delicado, entao se voce nao tiver cuidado, pum! Vai explodir.";
			next;
			mes "[Sratos]";
			mes "De qualquer forma, se voce usar essa gema eterica, pode aprimorar a arma que esta usando.";
			mes "O desempenho melhora. Nos chamamos isso de aprimoramento de grau.";
			next;
			mes "[Sratos]";
			mes "Voce tambem, nao ficaria feliz se as armas que voce ama crescessem mais um passo?";
			mes "Nos somos as pessoas que auxiliam nisso. E um pouco arriscado, porem. ahahaha.";
			close;
	}

OnInit:
	setunittitle(getnpcid(0), "<Grade Enhancer>");
	end;
}
