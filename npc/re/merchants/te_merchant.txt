	mes "[Rental Items Manager]";

		mes "Voce ja tem itens demais para receber um novo equipamento de aluguel.";
		mes "Por favor, organize seus itens e tente novamente.";

	}

	mes "Selecione o item desejado.";
	next;
	for ( .@i = 1; .@i < getargcount(); .@i += 2 ) {
		if (rentalcountitem(getarg(.@i)) > isequippedcnt(getarg(.@i)))// don't count item equipped
			.@menu$ = .@menu$ + "^4d4dff"+ getarg(.@i+1) +" - Ja retirado^000000:";
		else
			.@menu$ = .@menu$ + getarg(.@i+1) + ":";
	}
	.@total_item = (getargcount() -1) /2;
	.@s = select( .@menu$ + ":Ver outros itens" ) -1;
	if (.@s <= .@total_item) {
		.@index = .@s *2 +1;
		mes "[Rental Items Manager]";
		if (rentalcountitem(getarg(.@index)) > isequippedcnt(getarg(.@index)))
			mes "Voce ja tem "+ getarg(.@index+1) +".";
		else {
			mes "Aqui esta "+ getarg(.@index+1) +".";
			mes "^4d4dffEsse item e valido pelas proximas 24 horas^000000 e depois disso ele expira.";
			rentitem getarg(.@index),86400;
		}
		mes "Voce precisa de mais alguma coisa?";
		next;
	}
	return;
