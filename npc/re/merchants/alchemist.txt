alde_alche,31,186,3	script	Craft Book Merchant#alde	883,{
	mes "[Craft Book Merchant]";
	mes "Bem-vindo.";
	mes "Estou aqui para vender";
	mes "^0000FFlivros de receitas especiais^000000 para Geneticistas.";
	next;
	mes "[Craft Book Merchant]";
	mes "Geneticistas tem algumas habilidades";
	mes "que precisam de livros de receitas para ativar.";
	next;
	mes "[Craft Book Merchant]";
	mes "Quanto mais livros de receitas voce tiver,";
	mes "mais habilidades voce podera usar.";
	next;
	mes "[Craft Book Merchant]";
	mes "Por favor, entenda que";
	mes "eu me especializo na venda desses livros especiais e";
	mes "eu ^FF0000nao vendo outros itens^000000.";
	next;
	mes "[Craft Book Merchant]";
	mes "Voce gostaria de dar uma olhada nos meus livros de receitas?";
	next;
	if(select("Claro.:Nao.") == 2) {
		mes "[Craft Book Merchant]";
		mes "Obrigado por visitar minha loja.";
		mes "Por favor, tenha cuidado em suas viagens.";
		close;
	}
	if (checkweight(1201,1) == 0) {
		mes "[Craft Book Merchant]";
		mes "Sinto muito, mas seu inventario esta quase cheio.";
		mes "Por favor, esvazie seu inventario primeiro.";
		close;
	}
	if (MaxWeight - Weight < 2500) {
		mes "[Craft Book Merchant]";
		mes "Por favor, esvazie seu inventario primeiro.";
		close;
	}
	mes "[Craft Book Merchant]";
	mes "Certo entao, aqui esta a lista de livros de receitas disponiveis.";
	next;
	switch(select("[Apple Bomb Craft Book] 100.000 zeny:[Pineapple Bomb Craft Book] 100.000 zeny:[Coconut Bomb Craft Book] 100.000 zeny:[Melon Bomb Craft Book] 100.000 zeny:[Banana Bomb Craft Book] 100.000 zeny:[Plant Gene Cultivation Method] 100.000 zeny:[Superior Potion Craft Manual] 100.000 zeny:[Mix Cooking Book] 100.000 zeny:[Health Improvement Research Book] 100.000 zeny:[Vigor Drink Recipe] 100.000 zeny:Fechar")){
	case 1: set .@item,6279; break;
	case 2: set .@item,6280; break;
	case 3: set .@item,6281; break;
	case 4: set .@item,6282; break;
	case 5: set .@item,6283; break;
	case 6: set .@item,6284; break;
	case 7: set .@item,6285; break;
	case 8: set .@item,11022; break;
	case 9: set .@item,11023; break;
	case 10: set .@item,11024; break;
	case 11:
		mes "[Craft Book Merchant]";
		mes "Obrigado pela preferencia.";
		mes "Por favor, volte novamente.";
		close;
	}
	mes "[Craft Book Merchant]";
	mes "^0000FF["+getitemname(.@item)+"] custa";
	mes "100.000 zeny.";
	mes "Quantos voce gostaria de comprar?";
	next;
	input .@amount;
	if (.@amount == 0) {
		mes "[Craft Book Merchant]";
		mes "Voce gostaria de ver outros livros?";
		close;
	}
	if (.@amount < 1 || .@amount > 99 ){
		mes "[Craft Book Merchant]";
		mes "Voce nao pode comprar mais de 100 por vez.";
		close;
	}
	mes "[Craft Book Merchant]";
	mes "Voce digitou "+.@amount+"x ^0000FF["+getitemname(.@item)+"]^000000 para serem comprados.";
	mes "Gostaria de continuar?";
	next;
	if (select("Sim:Nao") == 1) {
		set .@total, .@amount * 100000;
		if (Zeny < .@total) {
			mes "[Craft Book Merchant]";
			mes "Sinto muito, mas voce nao tem dinheiro suficiente.";
			close;
		}
		if (checkweight(.@item,.@amount) == 0){
			mes "[Craft Book Merchant]";
			mes "Nao parece que voce consegue carregar tudo isso.";
			mes "Por favor, verifique o espaco no seu inventario.";
			close;
		}
		mes "[Craft Book Merchant]";
		mes "Obrigado pela preferencia.";
		set Zeny,Zeny - .@total;
		getitem .@item,.@amount;
		close;
	}
	mes "[Craft Book Merchant]";
	mes "Por favor, tome seu tempo";
	mes "antes de tomar sua decisao.";
	close;
}
