ecl_in02,164,56,3	script	Receptionist#Laphine	835,{
	mes "[Receptionist]";
	mes "Bem-vindo a Clinica de Eclage.";
	next;
	switch (select("Salvar Localizacao:Descansar:Encerrar conversa")) {
	case 1:
		mes "[Receptionist]";
		mes "A clinica foi definida como seu ponto de retorno. Tome cuidado~";
		savepoint "ecl_in02",162,50,1,1;
		close;
	case 2:
		mes "[Receptionist]";
		mes "Restaure sua mente e corpo na nossa clinica. Ela e preenchida com a energia pura de Eclage.";
		mes "Cobramos 1 Splendide Coin normalmente, mas oferecemos um desconto especial para viajantes. Seu total e 5.000 zeny. ";
		next;
		switch (select("Pagar com Splendide Coin:Pagar com Zeny:Deixa pra la")) {
		case 1:
			if (!countitem(6081)) {
				mes "[Receptionist]";
				mes "Parece que voce nao tem moedas suficientes. Poderia verificar sua conta novamente, por favor?";
				close;
			}
			mes "[Receptionist]";
			mes "Fique a vontade~";
			close2;
			delitem 6081,1; //6081
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;
		case 2:
			if (Zeny < 5000) {
				mes "[Receptionist]";
				mes "Parece que voce nao tem recursos suficientes para prosseguir. Poderia verificar sua conta novamente?";
				close;
			}
			mes "[Receptionist]";
			mes "Fique a vontade~";
			close2;
			Zeny = Zeny - 5000;
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;
		case 3:
			close;
		}
	case 3:
		close;
	}
}
