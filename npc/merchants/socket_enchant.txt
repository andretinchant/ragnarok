function	script	Func_Socket	{
	.@item_id = getarg(0);
	.@zeny_req = getarg(4);
	setarray .@item_req[0], getarg(5), getarg(7,0);
	setarray .@req_amount[0], getarg(6), getarg(8,0);

	disable_items;
	mes "[Seiyablem]";
	mes "Voce quer adicionar um Slot a um "+ getitemname(.@item_id) +"?";
	mes "Certo, me traga ^FF0000"+ .@req_amount[0] +" "+ getitemname(.@item_req[0]) + (.@req_amount[0] > 1 && .@item_req[0] != 999 ? "s":"") +"^000000, "+ (.@item_req[1] != 0 && .@req_amount[1] != 0 ? ("^FF0000"+ .@req_amount[1] +" "+ getitemname(.@item_req[1]) + (.@req_amount[1] > 1 && .@item_req[1] != 999 ? "s":"") + "^000000 ") : "") + "e minha ^FF0000" + (.@zeny_req >= 1000 ? (.@zeny_req /1000 +",000") : .@zeny_req) +",000 zeny^000000 taxa de servico.";
	mes "Ah, e nao esqueca de trazer o "+ getitemname(.@item_id) +"!";
	next;
	mes "[Seiyablem]";
	if (getiteminfo(.@item_id, ITEMINFO_LOCATIONS) & EQP_HAND_R) // weapon
	{
		mes "Posso tentar adicionar um slot agora se voce tiver os itens e zeny necessarios.";
		mes "Porem, voce deve saber que existe uma chance de eu falhar.";
		mes "Portanto, preciso te dar um aviso justo...";
		next;
		mes "[Seiyablem]";
		mes "Se essa tentativa de adicionar um Slot a sua Arma falhar, a ^FF0000Arma^000000, ^FF0000e quaisquer Cartas compostas nela serao destruidas^000000.";
	}
	else // armor
	{
		mes "Se voce tiver todos os materiais necessarios, minha taxa de servico em zeny e a Armadura, podemos prosseguir com a tentativa de Adicao de Slot.";
		mes "Mas antes disso, devo te avisar do risco.";
		next;
		mes "[Seiyablem]";
		mes "Se essa tentativa de adicionar um Slot a sua Armadura falhar, a ^FF0000Armadura^000000, ^FF0000seus upgrades^000000 ^FF0000e quaisquer Cartas compostas nela serao destruidos^000000.";
	}
	mes "Voce ainda quer tentar adicionar um Slot?";
	next;
	switch(select("Tentar Adicao de Slot:Cancelar"))
	{
		case 1:
			if ((Zeny >= .@zeny_req*1000) && countitem(.@item_req[0]) >= .@req_amount[0] && countitem(getarg(7,512)) >= .@req_amount[1] && countitem(.@item_id) > 0)// if there's no getarg(7) and getarg(8) provided, check itemcount against 0 (we still need a valid item ID though)
			{
				mes "[Seiyablem]";
				mes "Certo entao, vamos comecar o trabalho!";
				mes "E melhor voce rezar por um resultado bem-sucedido.";
				next;
				set .@a, rand(1,100);
				if ((.@a > getarg(2)) && (.@a < getarg(3)))
				{
					specialeffect getarg(3) == 51 ? EF_LORD : EF_SANCTUARY; // only S Class items have Lord of Vermillion effect
					mes "[Seiyablem]";
					mes "Otimo, parece ter sido um sucesso.";
					mes "Ficou muito bem feito. Parabens!";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[Seiyablem]";
					mes "Ate a proxima, amigo!";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[Seiyablem]";
					mes "Uah! ...Sinto muito, falhou.";
					mes "Porem, sou completamente inocente.";
					mes "Isso e sua sorte, e e destino de Deus, ta?";
					mes "Nao fique tao desapontado, e tente da proxima vez.";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[Seiyablem]";
					mes "Te desejo boa sorte da proxima vez!";
					close;
				}
			}
			else
			{
				mes "[Seiyablem]";
				mes "Gostaria de prosseguir com essa tentativa de Adicao de Slot, mas esta faltando algumas coisas.";
				mes "Tem certeza que tem o equipamento, materiais necessarios e o zeny?";
				close;
			}
		case 2:
			mes "[Seiyablem]";
			mes "Precisa de um tempo pra pensar, ne?";
			mes "Certo, eu entendo.";
			mes "So lembre-se que a vida nao tem graca se voce esta sempre jogando pelo seguro~";
			close;
	}
}
