//callfunc "F_CoolEventCorp","<menu>","<location>","<save map>",<x>,<y>,"<warp map>";
function	script	F_CoolEventCorp	{
	cutin "zonda_01",2;
	mes "[Cool Event Corp. Staff]";
	mes "Bem-vindo a Cool Event Corp.";
	mes "Nossa equipe esta sempre trabalhando";
	mes "para superar suas expectativas";
	mes "de servico de qualidade. Entao como";
	mes "posso ajuda-lo hoje?";
	next;
	switch(select(getarg(0))) {
	case 1:
		mes "[Cool Event Corp. Staff]";
		mes "Seu Ponto de Retorno";
		mes "foi salvo aqui";
		mes getarg(1)+".";
		mes "Obrigado por usar o";
		mes "servico da Cool Event Corp.~";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close2;
		break;
	case 2:
		mes "[Cool Event Corp. Staff]";
		if(!callfunc("F_CanOpenStorage")){
			mes "Desculpe, mas voce";
			mes "precisa da habilidade";
			mes "Basica de Novato nivel 6 para";
			mes "usar o Servico de Armazem.";
			close2;
			break;
		}
		if(Zeny < 40){
			mes "Desculpe, mas voce nao";
			mes "tem Zeny suficiente para usar";
			mes "o Servico de Armazem. Nossa";
			mes "taxa de acesso ao Armazem e de 40 Zeny.";
			close2;
			break;
		}
		set Zeny, Zeny - 40;
		mes "Deixe-me abrir seu";
		mes "armazem pessoal agora mesmo.";
		mes "Obrigado por apoiar a Cool";
		mes "Event Corp. usando nossos";
		mes "servicos. Tenha um bom dia~";
		callfunc("F_CheckKafCode");	//check your storage password, if set
		close2;
		openstorage;
		break;
	case 3:
		mes "Por favor escolha";
		mes "seu destino.";
		next;
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE))
			set .@cost,4400;
		else
			set .@cost,2200;
		.@destination$ = getarg(5);
		if (select(.@destination$ +" -> "+ .@cost +" z:Cancel") == 1) {
			if ((countitem(7060) != 0) || (Zeny >= .@cost)) {
				if (countitem(7060) != 0) {
					delitem 7060,1; //Warp_Free_Ticket
				}
				else {
					set Zeny,Zeny-.@cost;
				}
				if (.@destination$ == "Veins") warp "veins",205,101;
				else if (.@destination$ == "Rachel") warp "rachel",115,125;
				cutin "",255;
				end;
			}
			else {
				mes "[Cool Event Corp. Staff]";
				mes "Desculpe, mas voce nao tem";
				mes "zeny suficiente para o Servico";
				mes "de Teleporte. A taxa para teleportar";
				mes "para "+ .@destination$ +" e de "+.@cost+" zeny.";
			}
		}
		close2;
		break;
	case 4:
		mes "[Cool Event Corp. Staff]";
		if(BaseClass != Job_Merchant){
			mes "Desculpe, mas o";
			mes "servico de aluguel de Carrinho";
			if (checkre(0))
				mes "esta disponivel apenas para classes Mercador.";
			else {
				mes "esta disponivel apenas para Mercadores,";
				mes "Ferreiros, Mestre Ferreiros,";
				mes "Alquimistas e Criadores.";
			}
			close2;
			break;
		}
		if(getskilllv("MC_PUSHCART") == 0){
			mes "Voce so pode alugar um carrinho apos";
			mes "aprender a habilidade Carrinho.";
			close2;
			break;
		}
		if(checkcart() == 1){
			mes "Voce ja tem";
			mes "um Carrinho equipado.";
			mes "Infelizmente, nao podemos";
			mes "alugar mais de um para";
			mes "cada cliente por vez.";
			close2;
			break;
		}
		if(countitem(7061)){
			delitem 7061,1; //Cart_Free_Ticket
			setcart;
			close2;
			break;
		}
		mes "O aluguel de Carrinho";
		mes "e de 800 Zeny. Gostaria de";
		mes "alugar um Carrinho?";
		next;
		if(select("Alugar Carrinho:Cancelar") == 1){
			if(Zeny < 800){
				mes "[Cool Event Corp. Staff]";
				mes "Desculpe, mas voce";
				mes "nao tem";
				mes "Zeny suficiente para pagar o";
				mes "aluguel de Carrinho de 800 Zeny.";
			} else {
				set Zeny, Zeny - 800;
				setcart;
			}
		}
		close2;
		break;
	case 5:
		callfunc("F_SetKafCode","[Cool Event Corp. Staff]","Cool Event Corp.");
		//the function closes msg and removes cutins
	case 6:
	default:
		mes "[Cool Event Corp. Staff]";
		mes "A Cool Event Corp. esta sempre";
		mes "se esforcando para oferecer os melhores";
		mes "servicos para nossos clientes.";
		mes "Ajude-nos a ser os melhores";
		mes "nos dando suas opinioes";
		mes "e feedback sincero. Obrigado.";
		close2;
	}
	cutin "", 255;
	end;
}

		"Save:Use Storage::Rent a Pushcart:Storage Password Service:Cancel",
		"in the town of Lighthalzen","lighthalzen",95,240;
	end;
}
