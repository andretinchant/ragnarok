//=== Main Function ========================================================
//=   arg(0): Used to determine which welcome message to show.
//=   arg(1): Used to determine which menu to display.
//=   arg(2): Used to determine if the info menu is shown in F_KafInfo.
//=   arg(3): Cost to use Storage
//=   arg(4): Cost to Rent a Pushcart
//==========================================================================
function	script	F_Kafra	{
	callfunc "F_ClearGarbage"; // Clear outdated, unused variables
	.@welcome = getarg(0);
	.@menu_num = getarg(1);

	// Display Kafra Welcome Message
	switch(.@welcome) {
	default:// Default message (obsolete)
		mes "[Kafra Employee]";
		mes "Bem-vindo a";
		mes "Corporacao Kafra.";
		mes "Os servicos Kafra";
		mes "estao sempre ao seu lado.";
		mes "Como posso ajuda-lo?";
		break;
	case 1:// Niflheim specific message
		mes "[Kafra Employee]";
		mes "^666666B-beem-v-vindo";
		mes "a-a C-coorpoor-r-racao";
		mes "K-kaaafrrrra...^000000";
		break;
	case 2:// Guild Castle Kafra message
		mes "[Kafra Employee]";
		mes "Bem-vindo. ^ff0000" + getguildname( getcharid(2) ) + "^000000 Membro.";
		mes "A Corporacao Kafra ficara com voce onde quer que va.";
		break;
	case 3:// Amatsu specific message (obsolete)
		mes "[Kafra Employee]";
		mes "Entao, voce veio de uma terra distante para estudar nossa cultura, ou esta apenas passeando?";
		mes "Em qualquer caso, por que nao fica um pouco?";
		mes "O ar e eternamente carregado com o";
		mes "perfume de agradaveis flores silvestres.";
		break;
	case 4:// Luoyang and Ayothaya specific message (obsolete)
		mes "[Kafra Employee]";
		mes "Com nossas muitas";
		mes "localizacoes de servico Kafra, voce nunca";
		mes "esta longe de casa.";
		break;
	case 5:// NPC has it's own welcome message. (Display nothing)
		break;
	}
	next;
	if (.@welcome == 2)// Guild Kafra Menu override (free Teleport, free Storage, Free Cart)
		setarray .@K_Menu0$[0], "Usar Armazem", "Usar Armazem da Guilda", "Comprar Itens Kafra", "Alugar Carrinho", "Usar Servico de Teleporte", "Cancelar";
	else {
		switch(.@menu_num) {
			// Save and Storage only
			case 1:	setarray .@K_Menu0$[0],"Salvar","Usar Armazem","Comprar Itens Kafra","Cancelar"; break;
			// Storage only
			case 2:	setarray .@K_Menu0$[0],"Usar Armazem","Comprar Itens Kafra","Cancelar"; break;
			// No Teleport (Common)
			case 3: setarray .@K_Menu0$[0],"Salvar","Usar Armazem","Comprar Itens Kafra","Alugar Carrinho","Verificar Outras Informacoes","Cancelar"; break;
			// Case 4 is Einbroch no tele message.
			// No save, or teleport. (Common)
			case 5:	setarray .@K_Menu0$[0],"Usar Armazem","Comprar Itens Kafra","Alugar Carrinho","Verificar Outras Informacoes","Cancelar"; break;
			// Storage and Check Other Information only.
			case 6:	setarray .@K_Menu0$[0],"Usar Armazem","Comprar Itens Kafra","Verificar Outras Informacoes","Cancelar"; break;
			// Save, Storage, and Pushcart only (Kafra Warehouse)
			case 7:	setarray .@K_Menu0$[0],"Salvar","Usar Armazem","Comprar Itens Kafra","Alugar Carrinho","Cancelar"; break;
			// Save, Storage, Other Check information. (Turbo track)
			case 8: setarray .@K_Menu0$[0],"Salvar","Usar Armazem","Comprar Itens Kafra","Verificar Outras Informacoes","Cancelar"; break;
			// No Save (Rune Knight)
			case 9: setarray .@K_Menu0$[0],"Usar Armazem","Comprar Itens Kafra","Alugar Carrinho","Usar Servico de Teleporte","Verificar Outras Informacoes","Cancelar"; break;
			// Storage, Save, and Pushcart (Dewata, reorder of case 7)
			case 10: setarray .@K_Menu0$[0],"Usar Armazem","Salvar","Comprar Itens Kafra","Alugar Carrinho","Cancelar"; break;
			// Default message (obsolete)
			default: setarray .@K_Menu0$[0],"Salvar","Usar Armazem","Comprar Itens Kafra","Usar Servico de Teleporte","Alugar Carrinho","Verificar Outras Informacoes","Cancelar"; break;
		}
	}
	.@menu$ = implode(.@K_Menu0$,":");
	while (true) {
		.@j = select(.@menu$) - 1;
		if (.@K_Menu0$[.@j] == "Salvar")
			return;
		else if (.@K_Menu0$[.@j] == "Usar Armazem") {
			if (.@welcome == 2)// Do not charge for Guild Storage
				callfunc "F_KafStor",2,0,0;
			else
				callfunc "F_KafStor",0,getarg(3),.@welcome;
			next;
		}
		else if (.@K_Menu0$[.@j] == "Comprar Itens Kafra") {
			callfunc "F_KafBuyCard";
			next;
		}
		else if (.@K_Menu0$[.@j] == "Usar Servico de Teleporte") {
			if (.@menu_num != 4)// Display Einbroch "No Teleport Service" notice.
				callfunc "F_KafTele",.@welcome;
			else {
				mes "[Kafra Employee]";
				mes "Por causa do ^FF0000Acordo de";
				mes "Transporte Limitado^000000, a";
				mes "Corporacao Kafra nao pode";
				mes "fornecer Servicos de Teleporte";
				mes "na Republica de Schwarzwald.";
				next;
				mes "[Kafra Employee]";
				mes "Pedimos que por favor";
				mes "use o Servico de Aeronave";
				mes "em vez disso. Obrigado pela sua";
				mes "compreensao e cooperacao.";
			}
			next;
		}
		else if (.@K_Menu0$[.@j] == "Alugar Carrinho") {
			if (callfunc("F_KafCart",.@welcome,getarg(4)) == 1)
				next;
		}
		else if (.@K_Menu0$[.@j] == "Verificar Outras Informacoes") {
			callfunc "F_KafInfo",getarg(2);
			next;
		}
		else if (.@K_Menu0$[.@j] == "Cancelar") {
			callfunc "F_KafEnd",.@welcome,0;
			end;
		}
		else if (.@K_Menu0$[.@j] == "Usar Armazem da Guilda") {
			callfunc "F_KafStor",1,0;
			next;
		}
	}
}

//===================================================
// Kafra Card Purchase Function
//===================================================
function	script	F_KafBuyCard	{
	.@item_count = 1;
	.@item_id[0] = 12211;
	.@item_name$[0] = getitemname(12211);
	.@price[0] = 5000;
	.@price[1] = 47500;
	.@price[2] = 450000;
	.@amount[0] = 1;
	.@amount[1] = 10;
	.@amount[2] = 100;

	if (checkre(0)) {
		.@item_id[1] = 25464;
		.@item_name$[1] = "Passaporte";
		.@item_count = 2;
	}

	mes "[Kafra Employee]";
	mes "Posso vender itens de conveniencia";
	mes "da Corporacao Kafra para voce.";
	next;

	for (.@i = 0; .@i < .@item_count; .@i++)
		.@item_menu$ += .@item_name$[.@i] + ":";
	.@item_menu$ += "Cancelar";

	.@item_choice = select(.@item_menu$) - 1;
	if (.@item_choice >= .@item_count)
		return;

	.@selected_id = .@item_id[.@item_choice];
	.@selected_name$ = .@item_name$[.@item_choice];

	mes "[Kafra Employee]";
	mes "Qual pacote de ^0055FF" + .@selected_name$ + "^000000 voce deseja comprar?";
	next;

	.@pack_menu$ =
		"1 unidade -> " + .@price[0] + " zeny:" +
		"10 unidades -> " + .@price[1] + " zeny:" +
		"100 unidades -> " + .@price[2] + " zeny:" +
		"Cancelar";
	.@pack_choice = select(.@pack_menu$) - 1;

	if (.@pack_choice > 2)
		return;

	.@selected_amount = .@amount[.@pack_choice];
	.@selected_price = .@price[.@pack_choice];

	if (!checkweight(.@selected_id,.@selected_amount)) {
		mes "[Kafra Employee]";
		mes "Parece que voce esta";
		mes "com peso ou itens demais";
		mes "para carregar este pacote.";
		return;
	}

	if (Zeny < .@selected_price) {
		mes "[Kafra Employee]";
		mes "Desculpe, mas voce nao";
		mes "tem zeny suficiente.";
		mes "Esse pacote custa " + .@selected_price + " zeny.";
		return;
	}

	Zeny -= .@selected_price;
	getitem .@selected_id,.@selected_amount;

	mes "[Kafra Employee]";
	mes "Aqui esta seu pacote de";
	mes "^0055FF" + .@selected_amount + " " + .@selected_name$ + "^000000.";
	return;
}

//===================================================
// Storage Function
//===================================================
function	script	F_KafStor	{
	.@type = getarg(0);
	.@fee = getarg(1);

	// Unable to access Guild Storage (Busy)
	if (.@type == 1) {
		if (guildopenstorage()) {
			mes "[Kafra Employee]";
			mes "Desculpe, mas outro membro da guilda esta usando o armazem da guilda";
			mes "agora. Por favor, espere ate que essa pessoa termine.";
			close3;
		}
		cutin "", 255;
		close;
	}
	// Unable to access Normal Storage (Insufficient Basic Skills)
	if (!callfunc("F_CanOpenStorage")) {
		mes "[Kafra Employee]";
		// Niflheim Specific Message
		if (getarg(2) == 1) {
			mes "^666666D-d-deeesculllpe,";
			mes "v-voce e-e u-u-um";
			mes "Noovaaato... P-precisa";
			mes "H-habilidade b-basica n-nivel 6...^000000";
			return;
		}
		mes "Desculpe, mas voce";
		mes "precisa da habilidade";
		mes "Basica de Novato nivel 6 para";
		mes "usar o Servico de Armazem.";
		return;
	}
	// Accessing Normal Storage (Skipped if accessing Storage from Guild castle)
	if (.@type != 2) {
		// Consume "Free Ticket for Kafra Storage" if available.
		if (countitem(7059))
			delitem 7059,1;
		else {
			if (Zeny < .@fee) {
				mes "[Kafra Employee]";
				// Niflheim Specific Message
				if (getarg(2) == 1) {
					percentheal -50,-50;
					mes "^666666Zeeeeeny...";
					mes "M-mais z-zeny...!";
					mes "P-preciso de 150... zeny...";
					mes "Ergh! T-taking bl-blood~!^000000";
					return;
				}
				// Standard Message
				mes "Desculpe, mas voce nao";
				mes "tem zeny suficiente para usar";
				mes "o Servico de Armazem. Nossa";
				mes "taxa de acesso ao Armazem e de " + .@fee + " zeny.";
				return;
			}
			Zeny -= .@fee;
			RESRVPTS += (.@fee/5);
		}
	}

	// Surpress the "End" message if in guild castle.
	if (.@type != 2) {
		// Niflheim specific Message
		if (getarg(2) == 1) {
			percentheal 0,-10;
			mes "[Kafra Employee]";
			for ( .@i = 0; .@i < 5; .@i++ )
				mes "^666666Obrigado.. por... usar...^000000";
		}
		// Normal message
		else {
			mes "[Kafra Employee]";
			mes "Aqui, deixe-me abrir";
			mes "seu Armazem para voce.";
			mes "Obrigado por usar";
			mes "o Servico Kafra.";
		}
	}
	callfunc("F_CheckKafCode");	//check your storage password, if set
	close2;
	openstorage;
	cutin "", 255;
	end;
}

//===================================================
// Teleport Function
//===================================================
function	script	F_KafTele	{
	mes "[Kafra Employee]";
	mes "Por favor escolha";
	mes "seu destino.";
	next;
	.@j = select( implode(@wrpC$,":") ) - 1;
	if (@wrpC$[.@j] == "Cancelar")
		return;
	// Consume "Free Ticket for Kafra Transportation" if available.
	// Do not consume if in Guild castle
	if (countitem(7060) > 0 && getarg(0) != 2)
		delitem 7060,1;
	else {
		if (Zeny < @wrpP[.@j]) {
			mes "[Kafra Employee]";
			mes "Desculpe, mas voce nao tem";
			mes "zeny suficiente para o Servico";
			mes "de Teleporte. A taxa para teleportar";
			mes "para " + @wrpD$[.@j] + " e de " + @wrpP[.@j] + " zeny.";
			close3;
		}
		Zeny -= @wrpP[.@j];
		RESRVPTS += (@wrpP[.@j]/16);
	}
	cutin "", 255;
	if (@wrpD$[.@j] == "Al De Baran") warp "aldebaran",168,112;
	else if (@wrpD$[.@j] == "Alberta") warp "alberta", 117, 56;
	else if (@wrpD$[.@j] == "Comodo") warp "comodo", 209, 143;
	else if (@wrpD$[.@j] == "Comodo Pharos Beacon") warp "cmd_fild07", 127, 134;
	else if (@wrpD$[.@j] == "Geffen") warp "geffen", 120, 39;
	else if (@wrpD$[.@j] == "Izlude") {
		if (checkre(0) == true)
			warp "izlude", 128, 98;
		else
			warp "izlude", 91, 105;
	}
	else if (@wrpD$[.@j] == "Juno") warp "yuno", 158, 125;
	else if (@wrpD$[.@j] == "Mjolnir Dead Pit") warp "mjolnir_02", 99, 351;
	else if (@wrpD$[.@j] == "Morocc") warp "morocc", 156, 46;
	else if (@wrpD$[.@j] == "Orc Dungeon") warp "gef_fild10", 52, 326;
	else if (@wrpD$[.@j] == "Payon") warp "payon", 161, 58;
	else if (@wrpD$[.@j] == "Prontera") warp "prontera", 116, 72;
	else if (@wrpD$[.@j] == "Umbala") warp "umbala", 100, 154;
	end;
}


//===================================================
// Cart Function
//===================================================
function	script	F_KafCart	{
	// Ensure that the class wanting to rent a pushcart is a merchant
	if (BaseClass != Job_Merchant) {
		mes "[Kafra Employee]";
		mes "Desculpe, mas o";
		mes "servico de aluguel de Carrinho";
		if (checkre(0) == true)
			mes "esta disponivel apenas para classes Mercador.";
		else {
			mes "esta disponivel apenas para Mercadores,";
			mes "Ferreiros, Mestre Ferreiros,";
			mes "Alquimistas e Bioquimicos.";
		}
		return 1;
	}
	// Make sure the invoking character does not have a cart already
	else if (checkcart() == true) {
		mes "[Kafra Employee]";
		mes "Voce ja tem";
		mes "um Carrinho equipado.";
		mes "Infelizmente, nao podemos";
		mes "alugar mais de um para";
		mes "cada cliente por vez.";
		return 1;
	}
	// Check if the player has the Pushcart skill
	else if (getskilllv("MC_PUSHCART") == 0) {
		mes "[Kafra Employee]";
		mes "You can only rent a cart after learning the \"Push Cart\" skill.";
		return 1;
	}
	// Consume "Free Ticket for the Cart Service" if available.
	// Do not consume if in Guild castle.
	if (countitem(7061) > 0 && getarg(0) != 2)
		delitem 7061,1;
	else {
		.@rental_fee = getarg(1);
		mes "[Kafra Employee]";
		mes "O aluguel de Carrinho";
		mes "e de " + .@rental_fee + " zeny. Gostaria de";
		mes "alugar um Carrinho?";
		next;
		if (select("Alugar Carrinho.:Cancelar") == 2)
			return 0;
		if (Zeny < .@rental_fee) {
			mes "[Kafra Employee]";
			mes "Desculpe, mas voce";
			mes "nao tem";
			mes "zeny suficiente para pagar o aluguel";
			mes "aluguel de Carrinho de " + .@rental_fee + " zeny.";
			return 1;
		}
		Zeny -= .@rental_fee;
		RESRVPTS += 48;
	}
	setcart;
	return 1;
}

function	script	F_KafInfo	{
// Uncomment next line to block Kafra Storage Protection
//	.@block = 1;
	if (getarg(0) == 2)// Port Malaya, Rune Knight
		setarray .@menu$[0], "Check Special Reserve Points.", "", "", "Cancelar";

		setarray .@menu$[0], "Check Special Reserve Points.", "", "Kafra Employee Locations", "Cancelar";

	.@menu_list$ = implode(.@menu$,":");
	while (true) {
		.@j = select(.@menu_list$) - 1;
		switch(.@j) {
		case 0:// Check Special Reserve Points.
			mes "[Kafra Employee]";
			mes "Deixe-me ver...";
			mes "" + strcharinfo(0) + "...";
			mes "Ah, voce tem um total de";
			mes "" + RESRVPTS + " Pontos de Reserva Especial.";
			next;
			mes "[Kafra Employee]";
			mes "Voce pode trocar seus";
			mes "Pontos de Reserva Especial por";
			mes "recompensas na Sede Kafra em Al De Baran. Por favor use nossos";
			mes "servicos convenientes para ver os beneficios do nosso programa de recompensas.";
			if (getarg(0) == 1)
				return;
			next;
			break;
		case 1:// Storage Password Service
			callfunc("F_SetKafCode","[Kafra Employee]","Kafra Services");
		case 2:// Kafra Employee Locations
			for ( .@i = 0; .@i <= 3; .@i++ )
				viewpoint 1,@viewpX[.@i],@viewpY[.@i],(.@i+1),0xFF00FF;
			next;
			for ( .@i = 0; .@i <= 3; .@i++ )
				viewpoint 2,@viewpX[.@i],@viewpY[.@i],(.@i+1),0xFF00FF;
			break;
		default:
			return;
		}
	}
}

function	script	F_KafEnd	{
	.@type = getarg(0);
	mes "[Kafra Employee]";
	if (getarg(1) == true) {// Save
		mes "Seu Ponto de Retorno";
		mes "foi salvo aqui";
		mes "" + getarg(2) + ".";
		mes "Obrigado por usar";
		mes "os Servicos Kafra.";
	} else if (.@type == 0 || .@type == 5) {// Generic End
		mes "Nos, da Corporacao Kafra,";
		mes "estamos sempre nos esforcando para oferecer os melhores servicos. Esperamos atender suas necessidades de aventura e padroes de excelencia.";
	} else if (.@type == 1) {// Niflheim End
		percentheal 0,-25;
		mes "^666666Kaffffra n-nunca";
		mes "mooooooorre. No...";
		mes "S-seu laaaaaado~^000000";
	} else if (.@type == 2) {// RockRidge Save
		mes "Salvo.";
		mes "Obrigado pela sua preferencia.";
	}
	close3;
}

//===================================================
// Check Storage Password Function
//===================================================
function	script	F_CheckKafCode	{
	if (!#kafra_code)
		return;
	mes "Digite sua senha do armazem:";
	input .@code_;
	if (.@code_ != #kafra_code) {
		dispbottom "Wrong storage password.";
		close3;
	}
	@kafcode_try = 0;
	return;
}

// === Set / Change / Clear Storage Password Function ===
// getarg(0): NPC Name
// getarg(1): Company Name
//=======================================================
function	script	F_SetKafCode	{
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	if (#kafra_code == 0) {
		mes "" + .@comp_name$ + " orgulhosamente apresenta um novo servico:";
		mes "Protecao adicional do armazem com uma senha.";
		next;
		if (select( "Definir nova senha -> 5000z", "Cancelar" ) == 1)
			callsub( S_SET, .@npc_name$, .@comp_name$ );
		close3;
	}
	mes "Seu armazem esta protegido com uma senha. O que gostaria de fazer agora?";
	next;
	switch( select( "Alterar senha antiga -> 5000z", "Remover senha do armazem -> 1000z", "Cancelar" ) ) {
	case 1:
		mes .@npc_name$;
		mes "Primeiro, por favor digite sua ^0000FFsenha antiga^000000.";
		.@code = callfunc("F_EntKafCode");
		if (!.@code || .@code != #kafra_code) {
			mes "Senha incorreta. Voce nao pode definir uma nova senha.";
			emotion ET_SCRATCH;
			break;
		}
		next;
		callsub( S_SET, .@npc_name$, .@comp_name$ );
		break;
	case 2:
		mes .@npc_name$;
		mes "Por favor, digite sua senha antes da remocao.";
		.@code = callfunc("F_EntKafCode");
		if (!.@code) {
			mes "A senha nao foi removida.";
			emotion ET_SCRATCH;
			break;
		}
		next;
		mes .@npc_name$;
		if (Zeny < 1000) {
			mes "Voce nao tem zeny suficiente.";
			emotion ET_MONEY;
			break;
		}
		Zeny -= 1000;
		if (.@code == #kafra_code) {
			#kafra_code = 0;
			mes "Voce limpou sua senha do armazem com sucesso.";
			mes "Obrigado por usar " + .@comp_name$ + ".";
			emotion ET_THANKS;
		} else {
			mes "Senha incorreta. Nao devolveremos seus 1000z.";
			mes "Por favor, da proxima vez digite a senha correta.";
			emotion ET_SORRY;
		}
		break;
	case 3:
		break;
	}
	close3;

S_SET:
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	mes "Agora digite sua ^FF0000nova senha^000000 para proteger seu armazem de ladroes.";
	.@code = callfunc("F_EntKafCode");
	if (!.@code) {
		mes "A senha nao foi alterada.";
		emotion ET_SCRATCH;
		return;
	}
	next;
	mes .@npc_name$;
	if (Zeny < 5000) {
		mes "Voce nao tem zeny suficiente.";
		emotion ET_MONEY;
		return;
	}
	Zeny -= 5000;
	#kafra_code = .@code;
	mes "Voce protegeu seu armazem com uma senha secreta.";
	mes "Obrigado por usar " + .@comp_name$ + ".";
	emotion ET_THANKS;
	return;
}

//===================================================
// Basic Password Validation Function
//===================================================
function	script	F_EntKafCode	{
	mes "Digite um numero de 1000~10000000:";
	@kafcode_try += 1;
	if (@kafcode_try > 10) {
		@kafcode_try = 0;
		logmes "Tried to fit storage password.";
		return 0;
	}
	if (input(.@code_) == 1) {
		mes "Voce nao pode usar uma senha tao grande.";
		return 0;
	}
	if (.@code_ < 1000) {
		mes "Voce nao deveria usar uma senha tao curta.";
		return 0;
	}
	return .@code_;
}
