-	script	::Sailor_izlude	-1,{
	mes "[Sailor]";
	mes "Ei, pessoal!";
	mes "Atencao, atencao!";
	mes "Venham navegar ao vento";
	mes "num Navio fascinante!";
	mes "Depressa, depressa!";
	next;
	switch(select("Ilha Byalan -> 150 Zeny.:Marina de Alberta -> 500 Zeny.:Cancelar.")) {
	case 1:
		if (Zeny < 150) {
			mes "[Sailor]";
			mes "150 Zeny!";
			mes "Apenas 150 Zeny pra embarcar!";
			close;
		}
		set Zeny, Zeny - 150;
		warp "izlu2dun",107,50;
		end;
	case 2:
		if (Zeny < 500) {
			mes "[Sailor]";
			mes "500 Zeny!";
			mes "Apenas 500 Zeny pra embarcar!";
			close;
		}
		set Zeny, Zeny - 500;
		warp "alberta",188,169;
		end;
	case 3:
		close;
	}
}
