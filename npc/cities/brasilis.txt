-	script	::Crewman_bra2	-1,{
	mes "[Crewman]";
	mes "Ei, voce ja ouviu falar de um lugar chamado Brasilis?";
	mes "E uma cidade tropical que e quente como o deserto, mas tambem chuvosa. E um lugar muito misterioso.";
	next;
	mes "[Crewman]";
	mes "Recentemente encontramos uma nova rota maritima pra chegar la facilmente.";
	if (VIP_SCRIPT) {
		set .@cost, vip_status(VIP_STATUS_ACTIVE)?1000:10000;
		mes "Sao apenas 10.000 zeny pela ida e volta, e 1.000 pra VIP! Entao, quer ir?"; //custom translation (VIP)
	} else {
		set .@cost,10000;
		mes "Sao apenas 10.000 zeny pela ida e volta! Entao, quer ir?";
	}
	next;
	switch(select("Me leve pra Brasilis!:Vou ficar aqui.")) {
	case 1:
		if (Zeny < .@cost) {
			mes "[Crewman]";
			mes "Eu disse "+callfunc("F_InsertComma",.@cost)+" zeny.";
			close;
		}
		else {
			mes "[Crewman]";
			mes "Legal~!! Vamos la~!";
			set Zeny, Zeny-.@cost;
			close2;
			warp "brasilis",314,60;
			end;
		}
	case 2:
		mes "[Crewman]";
		mes "Bom, se algum dia tiver interesse, me avise que eu te levo ate la.";
		close;
	}
}

brasilis,137,77,5	script	Ice-Cream Maker	85,{
	mes "[Ice Cream Maker]";
	mes "Venha~venha~";
	mes "Sorvete e o lanche perfeito pra um dia quente~";
	mes "Custa apenas ^3355FF100 Zeny^000000~";
	mes "Sorvete~";
	mes "Compre seu Sorvete!";
	next;
	switch(select("Me da um!:Sorvete?:Cancelar.")) {
	case 1:
		mes "[Ice Cream Maker]";
		mes "Como tem muita gente querendo um sorvete geladinho, voce so pode pedir 5 de cada vez.";
		mes "Entao, quantos voce quer?";
		next;
		while(1) {
			input .@input; //,1,5;
			if (.@input == 0) {
				mes "[Ice Cream Maker]";
				mes "Nenhum?";
				mes "Ta bom, sai da frente, tenho clientes pra atender.";
				close;
			}
			else if ((.@input < 0) || (.@input > 5)) {
				mes "[Ice Cream Maker]";
				mes "Nossa.";
				mes "Voce pediu demais.";
				mes "Se comer mais de 5, vai ter que lutar com um monstro no estomago. Calma ai, parceiro.";
				next;
			}
			else
				break;
		}
		set .@icecream_hap,.@input*100;
		if (Zeny < .@icecream_hap) {
			mes "[Ice Cream Maker]";
			mes "Cara~! Voce nao tem dinheiro suficiente.";
			mes "Custa so ^3355FF100 Zeny^000000~ Serio!";
			close;
		}
		if (!checkweight(536,.@input)) {
			mes "[Ice Cream Maker]";
			mes "Parece que voce ta carregando coisa demais.";
			mes "Alivie sua mochila antes de comprar isso.";
			close;
		}
		set Zeny, Zeny-.@icecream_hap;
		getitem 536,.@input; //Ice_Cream
		close;
	case 2:
		mes "[Ice Cream Maker]";
		mes "'Sorvete e...";
		mes "Espera, voce nao sabe";
		mes "o que e Sorvete?";
		mes "Debaixo de que pedra voce";
		mes "tava vivendo?";
		next;
		mes "[Ice Cream Maker]";
		mes "Nem vou comecar a falar o quao estranho isso parece.";
		mes "Enfim, compre seu Sorvete aqui enquanto ta geladinho.";
		close;
	case 3:
		mes "[Ice Cream Maker]";
		mes "Nao perca a chance de comer o melhor Sorvete de toda a terra~!";
		close;
	}
}
