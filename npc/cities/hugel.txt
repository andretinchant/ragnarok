hu_in01,23,311,4	script	Party Supplies Shop	898,{
	mes "[Shopkeeper]";
	mes "Bem-vindo a loja de artigos";
	mes "para festa!";
	mes "Por que voce nao aproveita uns";
	mes "fogos de artificio espetaculares com seus";
	mes "amigos?";
	mes "Podemos te fornecer 5 deles";
	mes "por 500 zeny.";
	next;
	switch (select("Comprar:Cancelar")) {
		case 1:
			if (Zeny < 500) {
				mes "[Shopkeeper]";
				mes "Sinto muito, mas voce nao tem";
				mes "dinheiro suficiente~";
				close;
			}
			set Zeny,Zeny-500;
			getitem 12018,5;	// Fire_Cracker
			mes "[Shopkeeper]";
			mes "Aqui esta!";
			mes "Divirta-se com eles!";
			close;
		case 2:
			mes "[Shopkeeper]";
			mes "Obrigado, volte sempre.";
			close;
	}
}
