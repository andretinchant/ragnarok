-	script	::PR_Officer_moscovia	-1,{
	mes "[Moscovia P.R. Officer]";
	mes "Moscovia! Voce conhece Moscovia?";
	mes "O paraiso espalhado pelos mares infinitos...";
	mes "Bem-vindo a Moscovia!";
	mes "E aventureiro e mistico.";
	next;
	switch(select("Sobre Moscovia...:Ir para Moscovia.:Cancelar")) {
		case 1:
			mes "[Moscovia P.R. Officer]";
			mes "Moscovia e um belo reino";
			mes "numa ilha localizada ao norte de Rune-";
			mes "Midgarts.";
			next;
			mes "[Moscovia P.R. Officer]";
			mes "Tenho certeza de que voce ficara";
			mes "absolutamente fascinado";
			mes "pelo belo cenario de Moscovia";
			mes "e seu magnifico palacio.";
			next;
			mes "[Moscovia P.R. Officer]";
			mes "Agora que nosso longo inverno";
			mes "passou,";
			mes "fico feliz por poder te mostrar";
			mes "nossa bela cidade natal.";
			next;
			mes "[Moscovia P.R. Officer]";
			mes "Se voce quer visitar Moscovia,";
			mes "aproveite a chance agora!";
			mes "Vou te ajudar a ter uma otima viagem";
			mes "ate Moscovia!";
			close;

		case 2:
			mes "[Moscovia P.R. Officer]";
			mes "Ok entao, vamos comecar.";
			mes "Voce deve me pagar 10.000 zeny";
			mes "pra ir ate Moscovia.";
			mes "Mas quando voce voltar,";
			mes "nao precisa pagar.";
			next;
			mes "[Moscovia P.R. Officer]";
			mes "Podemos partir agora?";
			next;
			if (select("Vamos la!:Cancelar") == 2) {
				mes "[Moscovia P.R. Officer]";
				mes "Se voce esta muito ocupado agora,";
				mes "me avise quando quiser.";
				mes "Estou sempre pronto pra guiar qualquer um ate Moscovia.";
				close;
			}
			if (Zeny < 10000) {
				mes "[Moscovia P.R. Officer]";
				mes "Sinto muito, mas voce nao tem";
				mes "zeny suficiente agora.";
				mes "Voce precisa de 10.000 zeny";
				mes "pra ir ate Moscovia.";
				mes "Obrigado.";
				close;
			} else {
				mes "[Moscovia P.R. Officer]";
				mes "Ok entao, estamos partindo.";
				close2;
				set Zeny,Zeny - 10000;
				warp "moscovia",163,55;
				end;
			}
		case 3:
			mes "[Moscovia P.R. Officer]";
			mes "Se voce esta muito ocupado agora,";
			mes "me avise quando quiser.";
			mes "Estou sempre pronto pra guiar qualquer um ate Moscovia.";
			close;
	}
}
