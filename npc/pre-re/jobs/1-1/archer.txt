payon_in02,64,71,4	script	Archer Guildsman#archer	85,{
	if (Upper == 1) {
		if (Class == Job_Novice_High && (ADVJOB == Job_Sniper || ADVJOB == Job_Clown || ADVJOB == Job_Gypsy)) {
			mes "[Archer Guildsman]";
			mes "Ei, eu te conheco.";
			mes "Voce ja fez esse teste";
			mes "antes, nao foi?";
			next;
			mes "[Archer Guildsman]";
			mes "Ah, voce deve ter ido";
			mes "a Valhalla e renascido.";
			mes "Uau, isso e muito impressionante!";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Archer Guildsman]";
				mes "Err...";
				mes "E melhor voce aprender todas as Habilidades Basicas primeiro antes de se tornar um Arqueiro.";
				next;
				mes "[Archer Guildsman]";
				mes "Certo, ate mais.";
				close;
			}
			mes "[Archer Guildsman]";
			mes "Muito bem. Eu nao";
			mes "preciso dizer mais nada.";
			mes "Sei que voce sera um otimo Arqueiro...";
			next;
			skill "NV_TRICKDEAD",0,SKILL_PERM;
			jobchange Job_Archer_high;
			skill "AC_MAKINGARROW",1,SKILL_PERM;
			skill "AC_CHARGEARROW",1,SKILL_PERM;
			mes "[Archer Guildsman]";
			mes "Embora nao haja recompensa especial para voce dessa vez, espero que entenda. Cuide-se.";
			close;
		}
		else {
			mes "[Archer Guildsman]";
			mes "Oh...?";
			mes "Ei, o que voce";
			mes "esta fazendo aqui...?";
			next;
			mes "[Archer Guildsman]";
			mes "Posso ver que voce nao tem vocacao para ser Arqueiro. Parece que voce foi feito para";
			mes "outra coisa...";
			close;
		}
	}
	mes "[Archer Guildsman]";
	mes "Bom dia. Como posso ajuda-lo?";
	next;
	switch(select("Quero ser um Arqueiro.:Preciso saber os requisitos, por favor.:Nada, obrigado.")) {
	case 1:
		if (BaseJob == Job_Archer) {
			mes "[Archer Guildsman]";
			mes "Voce ja se tornou um Arqueiro...";
			close;
		}
		else if (BaseJob != Job_Archer && BaseJob != Job_Novice) {
			mes "[Archer Guildsman]";
			mes "Hmm...";
			mes "Voce nao parece nada com um Aprendiz...";
			next;
			mes "[Archer Guildsman]";
			mes "De qualquer forma, seja la o que voce for, nao pode escolher a classe de Arqueiro porque ja tem uma classe.";
			close;
		}
		if (job_archer_q == 0) {
			mes "[Archer Guildsman]";
			mes "Voce quer ser um Arqueiro?";
			mes "Se sim, voce precisa preencher este formulario de inscricao.";
			next;
			if (select("Inscrever-se.:Cancelar") == 1) {
				set job_archer_q,1;
				setquest 1004;
				mes "[Archer Guildsman]";
				mes "Ok, assine aqui. Certo, hm, vou te promover assim que voce cumprir os requisitos.";
				next;
				mes "[Archer Guildsman]";
				mes "Se voce acha que ja os cumpriu, podemos verificar agora.";
				mes "Voce esta pronto?";
				next;
				if (select("Sim, estou.:Nao, ainda nao.") == 1) {
					mes "[Archer Guildsman]";
					mes "Certo, deixe-me verificar.";
					next;
				}
				else {
					mes "[Archer Guildsman]";
					mes "Entendo. Fique a vontade se quiser olhar os requisitos.";
					close;
				}
			}
			else {
				mes "[Archer Guildsman]";
				mes "Bem, tudo bem.";
				mes "Ate a proxima.";
				close;
			}
		}
		mes "[Archer Guildsman]";
		mes "Voce e..." + strcharinfo(0) + "?";
		next;
		if (!callfunc("F_CanChangeJob")) {
			mes "[Archer Guildsman]";
			mes "Bem, voce nao esta no nivel de classe correto. Por favor, verifique os requisitos novamente.";
			next;
			mes "[Archer Guildsman]";
			mes "Seu nivel de classe deve ser pelo menos 10, e nao se esqueca que deve aprender todas as Habilidades Basicas. Quando fizer isso, volte.";
			close;
		}
		if (job_archer_q == 1) {
			set .@archer_item1,countitem(1066) * 5;
			set .@archer_item2,countitem(1067) * 3;
			set .@archer_item3,countitem(1068) * 2;
			set .@archer_item4,countitem(1019);
			set .@total_archer,.@archer_item1 + .@archer_item2 + .@archer_item3 + .@archer_item4;
			set .@total_archer2,(((.@archer_item2 + .@archer_item3) * 2) + .@archer_item4);
			mes "[Archer Guildsman]";
			mes "Excelente!";
			mes "Agora entao,";
			mes "vejamos...";
			next;
			mes "[Archer Guildsman]";
			mes "Vou avaliar o valor dos varios tipos de Troncos, necessarios para produzir um Arco, que voce trouxe.";
			next;
			if (countitem(1066) == 0 && countitem(1067) == 0 && countitem(1068) == 0 && countitem(1019) == 0) {
				mes "[Archer Guildsman]";
				mes "Hm...";
				mes "Infelizmente voce nao trouxe nenhum dos itens necessarios. Nao ha nada para eu avaliar.";
				close;
			}
			mes "[Archer Guildsman]";
			if (countitem(1066) != 0) mes " Grau S : " + countitem(1066) + " un, Nota: " + .@archer_item1 + " . ";
			if (countitem(1067) != 0) mes " Grau A : " + countitem(1067) + " un, Nota : " + .@archer_item2 + " . ";
			if (countitem(1068) != 0) mes " Grau B : " + countitem(1068) + " un, Nota : " + .@archer_item3 + " . ";
			if (countitem(1019) != 0) mes " Grau C : " + countitem(1019) + " un, Nota : " + .@archer_item4 + " . ";
			if (.@total_archer < 25) {
				mes "Notas Totais: ^FF0000" + .@total_archer + "^000000 / 40";
				next;
				mes "[Archer Guildsman]";
				mes "Menos de 25!? Voce precisa ter uma nota de pelo menos 25! Vamos, se esforce mais!";
				close;
			}
			else {
				mes "Notas Totais: ^0000FF" + .@total_archer + "^000000 / 40";
				next;
				mes "[Archer Guildsman]";
				if (.@total_archer > 40) {
					mes "Uau! Mais de 40!";
					mes "Excelente! Parabens!";
				}
				else if (.@total_archer > 30) {
					mes "Mais de 30! Bom trabalho!";
					mes "Parabens!";
				}
				else {
					mes "*Suspiro* Bem, voce mal passou... De qualquer forma, bom trabalho.";
				}
			}
			next;
			mes "[Archer Guildsman]";
			mes "Vou transferir esses Troncos para o nosso Departamento de Producao de Arcos. Agora que voce cumpriu os requisitos, deixe-me te promover agora mesmo!";
			if (countitem(1066) != 0) delitem 1066,countitem(1066); //Tree_Of_Archer_1
			if (countitem(1067) != 0) delitem 1067,countitem(1067); //Tree_Of_Archer_2
			if (countitem(1068) != 0) delitem 1068,countitem(1068); //Tree_Of_Archer_3
			if (countitem(1019) != 0) delitem 1019,countitem(1019); //Wooden_Block
		}
		next;
		callfunc "Job_Change",Job_Archer;
		callfunc "F_ClearJobVar";
		completequest 1004;
		mes "[Archer Guildsman]";
		mes "Parabens!";
		mes "Voce agora e um Arqueiro!";
		next;
		mes "[Archer Guildsman]";
		mes "Claro, esperamos que voce ajude a contribuir para o futuro da Guilda dos Arqueiros com seus esforcos.";
		next;
		mes "[Archer Guildsman]";
		mes "Ah, seu arco chegou do Departamento de Producao de Arcos. Aqui, pegue! E seu~";
		getitem 1702,1; //Bow_
		getitem 1750,.@total_archer2; //Arrow
		next;
		mes "[Archer Guildsman]";
		mes "Agora, va em frente. Cace com orgulho, sabendo que foi treinado por um dos melhores!";
		close;
	case 2:
		mes "[Archer Guildsman]";
		mes "Vou explicar os requisitos para ser um Arqueiro.";
		if (BaseJob != Job_Novice) {
			if (BaseJob == Job_Archer) {
				next;
				mes "[Archer Guildsman]";
				mes "Mas...";
				mes "Voce ja e um Arqueiro. Voce deveria saber disso...";
			}
			else {
				next;
				mes "[Archer Guildsman]";
				mes "Espere um segundo. Voce ja escolheu outra classe. Voce nao precisa saber disso~";
			}
			mes "Entao...Sim...nao ha motivo real para te dizer os requisitos...";
		}
		next;
		mes "[Archer Guildsman]";
		mes "Primeiro de tudo, voce precisa ter o Nivel de Classe 9 como Aprendiz, e saber todas as Habilidades Basicas.";
		next;
		mes "[Archer Guildsman]";
		mes "Um Arqueiro precisa de concentracao e reflexos extremamente altos, entao nao aceitamos aqueles que tem pouca paciencia.";
		next;
		mes "[Archer Guildsman]";
		mes "Voce tambem tem que coletar ^FF0000Troncos^000000. Existem 4 tipos diferentes de Troncos, cada um de qualidade diferente. Voce recebera diferentes notas pelos seus Troncos, dependendo da qualidade.";
		next;
		mes "[Archer Guildsman]";
		mes "Para se tornar um Arqueiro, voce deve receber uma nota de pelo menos ^0000FF25^000000 pontos de 40. Voce pode conseguir Troncos do 'Willow', a arvore. Mas cuidado. Eles podem ser monstros dificeis.";
		close;
	case 3:
		close;
	}
}
