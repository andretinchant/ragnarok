	if (checkquest(60179) != -1) {
		if (checkquest(60179,HUNTING) != 2 ) {
			mes "[Mantis Researcher]";
			mes "Voce ja terminou de cacar os 50 Mantis?";
			next;
			switch(select("Sim:Nao:Quero desistir")) {
			case 1:
				mes "[Mantis Researcher]";
				mes "Hmm, acho que voce ainda nao cacou 50 yet.";
				close;
			case 2:
				mes "[Mantis Researcher]";
				mes "Lembre-se, preciso de ajuda cacando Mantis entao va e cace 50 deles.";
				close;
			case 3:
				mes "[Mantis Researcher]";
				mes "Tem certeza que quer parar de cacar?";
				mes "Todo o progresso que voce fez sera apagado";
				next;
				switch(select("Sim:Nao")) {
				case 1:
					mes "[Mantis Researcher]";
					mes "Tudo bem entao, volte aqui se mudar de ideia.";
					erasequest 60179;
					close;
				case 2:
					mes "[Mantis Researcher]";
					mes "Por favor, mate 50 Mantis in order to make this a Place were people can go to and gather Experience.";
					close;
				}
			}
		}
		mes "[Mantis Researcher]";
		mes "Incrivel, voce fez isso rapidamente.";
		getexp 18818,7527;
		erasequest 60179;
		close;
	}
	if ((BaseLevel > 34) && (BaseLevel < 71)) {
		mes "[Mantis Researcher]";
		mes "Que otimo!";
		mes "Voce esta aqui para me ajudar, certo?";
		mes "Se nao, nao sei o que fazer com os monstros neste lugar!";
		next;
		switch(select("Claro:Nao")) {
		case 1:
			mes "[Mantis Researcher]";
			mes "Otimo! Eu sabia no momento que te vi que podia contar com voce!";
			mes "Preciso que voce reduza a quantidade de monstros que estao causando caos neste mapa!";
			mes "Ha muitos Mantis por aqui, o que torna impossivel para jogadores de nivel mais baixo ganhar boa experiencia aqui.";
			next;
			mes "[Mantis Researcher]";
			mes "Entao, voce ainda gostaria de me ajudar?";
			next;
			switch(select("Sure:Cancelar")) {
			case 1:
				if (BaseLevel > 70) {
					mes "[Mantis Researcher]";
					mes "Obrigado pela sua grande ajuda com os Mantis!";
					mes "Atualmente a situacao esta estavel e nao preciso mais da sua ajuda para mata-los.";
					close;
				}
				mes "[Mantis Researcher]";
				mes "Mate 50 desses Mantis e me avise quando terminar. Vou te recompensar pelos seus esforcos.";
				setquest 60179;
				close;
			case 2:
				mes "[Mantis Researcher]";
				mes "Se mudar de ideia, por favor volte.";
				close;
			}
		case 2:
			mes "[Mantis Researcher]";
			mes "Se mudar de ideia, por favor volte.";
			close;
		}
	}
	mes "[Mantis Researcher]";
	mes "Que bom, que bom!";
	next;
	mes "[Mantis Researcher]";
	mes "Desculpe, nao posso falar agora, estou sofrendo muito, esses malditos Mantis vao me matar.";

}
