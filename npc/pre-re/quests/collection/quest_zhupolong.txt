		if (checkquest(60182,HUNTING) != 2 ) {
			mes "[Dragon Hunter]";
			mes "Voce ja terminou de cacar os 50 Zhu Po Long?";
			next;
			switch(select("Sim:Nao:Quero desistir")) {
			case 1:
				mes "[Dragon Hunter]";
				mes "Hmm, acho que voce ainda nao cacou 50 yet.";
				close;
			case 2:
				mes "[Dragon Hunter]";
				mes "Lembre-se, preciso de ajuda cacando Zhu Po Long entao va e cace 50 deles.";
				close;
			case 3:
				mes "[Dragon Hunter]";
				mes "Tem certeza que quer parar de cacar?";
				mes "Todo o progresso que voce fez sera apagado";
				next;
				switch(select("Sim:Nao")) {
				case 1:
					mes "[Dragon Hunter]";
					mes "Tudo bem entao, volte aqui se mudar de ideia.";
					erasequest 60182;
					close;
				case 2:
					mes "[Dragon Hunter]";
					mes "Por favor, mate 50 Zhu Po Long in order to make this a Place were people can go to and gather Experience.";
					close;
				}
			}
		}
		mes "[Dragon Hunter]";
		mes "Incrivel, voce fez isso rapidamente.";
		getexp 151300,60520;
		erasequest 60182;
		close;
	}
	if ((BaseLevel > 49) && (BaseLevel < 91)) {
		mes "[Dragon Hunter]";
		mes "Que otimo!";
		mes "Voce esta aqui para me ajudar, certo?";
		mes "Se nao, nao sei o que fazer com os monstros neste lugar!";
		next;
		switch(select("Claro:Nao")) {
		case 1:
			mes "[Dragon Hunter]";
			mes "Otimo! Eu sabia no momento que te vi que podia contar com voce!";
			mes "Preciso que voce reduza a quantidade de monstros que estao causando caos neste mapa!";
			mes "Ha muitos Zhu Po Long por aqui, o que torna impossivel para jogadores de nivel mais baixo ganhar boa experiencia aqui.";
			next;
			mes "[Dragon Hunter]";
			mes "Entao, voce ainda gostaria de me ajudar?";
			next;
			switch(select("Sure:Cancelar")) {
			case 1:
				if (BaseLevel > 90) {
					mes "[Dragon Hunter]";
					mes "Obrigado pela sua grande ajuda com os Zhu Po Long!";
					mes "Atualmente a situacao esta estavel e nao preciso mais da sua ajuda para mata-los.";
					close;
				}
				mes "[Dragon Hunter]";
				mes "Mate 50 desses Zhu Po Long e me avise quando terminar. Vou te recompensar pelos seus esforcos.";
				setquest 60182;
				close;
			case 2:
				mes "[Dragon Hunter]";
				mes "Se mudar de ideia, por favor volte.";
				close;
			}
		case 2:
			mes "[Dragon Hunter]";
			mes "Se mudar de ideia, por favor volte.";
			close;
		}
	}
	mes "[Dragon Hunter]";
	mes "Que bom, que bom!";
	next;
	mes "[Dragon Hunter]";
	mes "Desculpe, nao posso falar agora, estou sofrendo muito, esses malditos Zhu Po Long vao me matar.";
