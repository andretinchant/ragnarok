que_ng,187,149,3	script	Magazine Dealer Kenny::mdk	4_M_01,{
	mes "[Kenny]";
	if (BaseClass == Job_Gunslinger) {
		mes "Bem-vindo a minha Loja de Magazines.";
		mes "Como voce deve saber, grandes quantidades";
		mes "de balas podem ser carregadas mais";
		mes "facilmente quando estao em Magazines. Agora, posso te interessar em";
		mes "algo em particular?";
		next;
		switch(select("Wind Sphere Pack:Shadow Sphere Pack:Poison Sphere Pack:Water Sphere Pack:Fire Sphere Pack:Cartridge:Blood Cartridge:Silver Cartridge:Cancelar")) {
		case 1: callsub S_Casing,13204,12144;// <bullet id>, <pack id>
		case 2: callsub S_Casing,13206,12145;
		case 3: callsub S_Casing,13205,12146;
		case 4: callsub S_Casing,13207,12147;
		case 5: callsub S_Casing,13203,12148;
		case 6: callsub S_Casing,13200,12149;
		case 7: callsub S_Casing,13202,12150;
		case 8: callsub S_Casing,13201,12151;
		case 9:
			mes "[Kenny]";
			mes "Bom, se voce algum dia achar";
			mes "que tem balas demais,";
			mes "venha me procurar.";
			mes "E uma boa ideia guardar";
			mes "balas nos meus Magazines.";
			close;
		}
	}
	mes "Bem-vindo a minha loja.";
	mes "Aqui, eu forneco Magazines";
	mes "e Cartridges para Gunslingers.";
	mes "Desculpe, mas nao parece que";
	mes "meus servicos seriam de alguma";
	mes "utilidade para voce, aventureiro.";
	next;
	mes "[Kenny]";
	mes "Eh, mas se voce conhecer";
	mes "algum Gunslinger, mande";
	mes "eles pra ca. Nunca se pode";
	mes "ter balas demais.";
	close;

S_Casing:
	.@bullet_id = getarg(0);
	.@pack_id = getarg(1);

	mes "[Kenny]";
	mes "Agora, voce pode trocar";
	mes callfunc("F_InsertPlural",500,getitemname(.@bullet_id));
	mes "e 500 zeny por 1 "+getitemname(.@pack_id)+", entao certifique-se";
	mes "de que voce tem balas";
	mes "e zeny suficientes para essa troca.";
	next;
	mes "[Kenny]";
	mes "Voce pode trocar "+callfunc("F_InsertPlural",500,getitemname(.@bullet_id))+" e 500 zeny";
	mes "por 1 "+getitemname(.@pack_id)+".";
	next;
	mes "[Kenny]";
	mes "Lembre-se que eu nao posso te dar";
	mes "mais de 50 Magazines";
	mes "de uma vez. Agora, por favor, insira";
	mes "o numero de Magazines que voce";
	mes "quer receber. Se quiser cancelar, apenas digite ''0.''";
	next;
	input .@amount;
	mes "[Kenny]";
	if (.@amount > 50 || .@amount == 0) {
		mes "Ei, eu nao posso te dar";
		mes "tantos Magazines.";
		mes "Por favor, tente novamente e";
		mes "insira um numero nao";
		mes "maior que 50.";
	}
	else if (countitem(.@bullet_id) >= (500*.@amount)) {
		if (Zeny >= (500*.@amount)) {
			if (checkweight(.@pack_id,.@amount) == 0) {
				mes "Ei, voce tem muita";
				mes "tralha enfiada no seu";
				mes "Inventario. Libere um pouco de";
				mes "espaco, e depois volte";
				mes "para trocar suas balas por";
				mes "alguns Magazines depois, ta bom?";
			} else {
				mes "Certo, aqui estao";
				mes "seus Magazines. Obrigado";
				mes "por visitar minha loja, e";
				mes "espero que voce use toda";
				mes "sua municao com sabedoria.";
				Zeny = Zeny -500*.@amount;
				delitem .@bullet_id,(500*.@amount);
				getitem .@pack_id,.@amount;
			}
		} else {
			mes "Desculpe, mas voce nao tem";
			mes "zeny suficiente para";
			mes "essa troca de Magazine.";
			mes "Volte a minha loja";
			mes "depois que voce juntar";
			mes "mais dinheiro.";
		}
	} else {
		mes "Desculpe, mas voce nao tem";
		mes "balas suficientes para";
		mes "essa troca de Magazine.";
		mes "Talvez voce devesse verificar";
		mes "seu Inventario, e";
		mes "depois voltar mais tarde.";
	}
	close;
}
