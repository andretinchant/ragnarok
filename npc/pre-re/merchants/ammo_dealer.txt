alberta,118,160,3	script	Bullet Dealer Tony#alb::bdt	4_M_04,{
	mes "[Tony]";
	if (BaseClass == Job_Gunslinger) {
		mes "Eu sou Tony, o Vendedor de Balas.";
		mes "Venha me procurar sempre que estiver";
		mes "com pouca municao. Apenas me traga";
		mes "os materiais, e eu farei";
		mes "as balas que voce precisa.";
		next;
		switch(select("Poison Sphere:Flare Sphere:Lighting Sphere:Blind Sphere:Freezing Sphere:Cancelar")) {
		case 1: callsub S_BulletTrade,937,10,13205;// <item required>, <amount req>, <bullet id>
		case 2: callsub S_BulletTrade,7097,2,13203;
		case 3: callsub S_BulletTrade,7053,3,13204;
		case 4: callsub S_BulletTrade,1024,5,13206;
		case 5: callsub S_BulletTrade,7054,2,13207;
		case 6:
			mes "[Tony]";
			mes "Mudou de ideia?";
			mes "Bom, se voce algum dia precisar";
			mes "de balas, eu estarei bem";
			mes "aqui. Volte quando";
			mes "achar que vai precisar de mais";
			mes "municao, Gunslinger.";
			close;
		}
	}
	mes "Ei, eu sou Tony. Eu sou o";
	mes "responsavel por distribuir";
	mes "e fazer balas para";
	mes "Gunslingers. E assim que";
	mes "nossa guilda gosta";
	mes "de fazer as coisas.";
	next;
	mes "[Tony]";
	mes "Me desculpe se voce veio";
	mes "aqui para comprar balas.";
	mes "Eu so posso fazer negocios com";
	mes "Gunslingers de verdade.";
	close;

S_BulletTrade:
	.@required_id = getarg(0);
	.@required_amount = getarg(1);
	.@bullet_id = getarg(2);

	mes "[Tony]";
	mes "Para cada conjunto de";
	mes callfunc("F_InsertPlural",30,getitemname(.@bullet_id))+",";
	mes "voce deve me dar";
	mes "1 Emveretarcon,";
	mes "1 Phracon, e";
	mes callfunc("F_InsertPlural",.@required_amount,getitemname(.@required_id))+".";
	next;
	mes "[Tony]";
	mes "Lembre-se que eu posso dar";
	mes "no maximo 500 conjuntos de";
	mes "30 balas por vez. Por favor,";
	mes "insira o numero de conjuntos de balas";
	mes "que voce quer. Se quiser";
	mes "cancelar, apenas digite ''0.''";
	next;
	input .@amount;
	mes "[Tony]";
	if (.@amount < 1 || .@amount > 500) {
		mes "Ei, eu nao posso te dar";
		mes "tantas balas. Nao";
		mes "esqueca de inserir um numero";
		mes "que nao seja maior que 500";
		mes "se voce quiser trocar seus";
		mes "itens por algumas balas.";
	}
	else if (countitem(1010) >= .@amount && countitem(1011) >= .@amount && countitem(.@required_id) >= (.@amount*.@required_amount)) {
		if (checkweight(.@bullet_id,.@amount * 30) == 0) {
			mes "Hm? Seu Inventario nao";
			mes "tem espaco suficiente para";
			mes "tantas balas. Volte depois";
			mes "quando tiver mais espaco";
			mes "disponivel. Tente colocar algumas";
			mes "coisas no Armazem Kafra.";
		} else {
			mes "Otimo, tudo parece";
			mes "estar em ordem. Deixe eu pegar";
			mes "esses materiais, e aqui estao";
			mes "suas balas. E um prazer";
			mes "fazer negocios com voce~";
			delitem 1010,.@amount; //Phracon
			delitem 1011,.@amount; //Emveretarcon
			delitem .@required_id,.@amount * .@required_amount;
			getitem .@bullet_id,.@amount * 30;
		}
	} else {
		mes "Hm. Parece que voce";
		mes "nao tem materiais suficientes";
		mes "para tantas balas. Bom,";
		mes "nao tem problema. E so";
		mes "voltar depois de juntar tudo";
		mes "que voce precisa, ta bom?";
	}
	close;
}

que_ng,187,156,3	duplicate(bdt)	Bullet Dealer Tony#ng	86
