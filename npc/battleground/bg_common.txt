		set .@menu$, .@menu$+getitemname(.@Items[.@i])+":";
	set .@i, (select(.@menu$)-1)*2;
	mes "[Erundek]";
	mes "You chose ^3131FF"+getitemname(.@Items[.@i])+"^000000.";
	switch(.@Items[.@i]) {
		case 2720: mes "Este item e apenas para classes Espadachim e Mestre Taekwon."; break;
		case 2721: mes "Este item e apenas para classe Gatuno."; break;
		case 2722: mes "Este item e apenas para classe Acolito."; break;
		case 2723: mes "Este item e apenas para classe Mago."; break;
		case 2724: mes "Este item e apenas para classe Arqueiro."; break;
		case 2725: mes "Este item e apenas para classe Mercador."; break;
		case 2733: mes "Este item e apenas para Pistoleiro."; break;
		default: break;
	}
	mes "You can exchange for this item with ^FF0000"+.@Items[.@i+1]+" "+getitemname(7828)+" or "+.@Items[.@i+1]+" "+getitemname(7829)+"^000000.";
	mes "Gostaria de trocar?";
	next;
	switch(select("Nao trocar:Trocar")) {
	case 1:
		mes "[Erundek]";
		mes "Precisa de mais tempo para verificar os itens?";
		break;
	case 2:
		mes "[Erundek]";
		mes "Qual Insignia voce quer trocar?";
		mes "You need ^3131FF"+.@Items[.@i+1]+" Badges^000000 to exchange.";
		next;
		if (.@item[0] < 12269 || .@item[0] > 12273 ) {
			mes "[Erundek]";
			mes "Lembre-se, Itens de Recompensa do Campo de Batalha sao ^FF0000vinculados ao personagem^000000. Tem certeza que quer este item?";
			next;
		}
		set .@j, select("Insignia de Bravura:Insignia de Valor:Cancelar");
		mes "[Erundek]";
		if (.@j == 3) {
			mes "Voce cancelou a troca.";
			break;
		}
		set .@cost, ((.@j==1)?7828:7829);
		if (countitem(.@cost) >= .@Items[.@i+1]) {
			mes "Obrigado pela troca.";
			delitem .@cost, .@Items[.@i+1];
			getitem .@Items[.@i],1;
		}
		else mes "Voce nao tem "+getitemname(.@cost)+" suficientes.";
		break;
	}
	close;
}

// Badge Reward Function
//============================================================
function	script	F_BG_Badge	{
	.@team$ = getarg(1);
	.@bg_type$ = getarg(2);

	if (.@bg_type$ == "Tierra") {
		set .@badge,7828; //BF_Badge1
		set .@amount_win,3;
		set .@amount_lose,1;
	} else if (.@bg_type$ == "Flavius") {
		set .@badge,7829; //BF_Badge2
		set .@amount_win,9;
		set .@amount_lose,3;
	} else
		return;

	if (getarg(0) == 1) {
		set .@amount, .@amount_win;
		if (.@team$ == "Guillaume") {
			mes "[Axl Rose]";
			mes "Guillaume abencoado!";
		} else if (.@team$ == "Croix") {
			mes "[Swandery]";
			mes "Croix abencoado!";
		}
		mes "Vamos aproveitar nossa gloriosa vitoria!";
		mes strcharinfo(0)+", e um sinal que reflete a vitoria.";
	} else {
		set .@amount, .@amount_lose;
		if (.@team$ == "Guillaume") {
			mes "[Axl Rose]";
			mes "Voce perdeu, mas foi dedicado a esta batalha.";
			mes "Esta e uma recompensa pela sua grande dedicacao por Guillaume Marollo!";
			mes "Apenas tome esta derrota como licao, e na proxima vez voce com certeza vencera.";
		} else if (.@team$ == "Croix") {
			mes "[Swandery]";
			mes "Oh, "+strcharinfo(0)+" Nao fique triste.";
			mes "Mesmo que nao tenhamos vencido, demos o nosso melhor.";
			mes "Este e um presente Real de Croix, por favor nao esqueca esta batalha. Venceremos a proxima.";
		}
	}
	close2;

	// If the VIP system is enabled, VIP players receive an extra 2 badges.
	if (VIP_SCRIPT && vip_status(VIP_STATUS_ACTIVE))
		set .@amount, .@amount+2;

	set .@medal_gap, 500 - countitem(.@badge);
	if (.@medal_gap >= .@amount)
		getitem .@badge,.@amount;
	else
		getitem .@badge,.@medal_gap;
	return;
}
