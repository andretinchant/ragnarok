//===== Description: =========================================
//= Removes cards from equipped items.
//===== Additional Comments: =================================
//= 1.0 First version. [TyrNemesis^]
//= 1.2 Optimized and fixed getequipname menu. [Kisuka]
//= 1.2a Added 'disable_items' command. [Euphy]

prt_in,28,73,4	script	Wise Old Woman#eAcustom	78,{

	set .zenycost,200000;    // base cost of the card remover services (in Zeny)
	set .percardcost,25000;  // cost per card of the card remover services (in Zeny)
	set .faildestroy,1;      // should the card remover have a chance of failure that destroys items? (1=yes, 0=no)

	disable_items;
	mes "[Wise Old Woman]";
	mes "Bom dia, jovem. Eu tenho o poder de remover cartas que voce compos no seu equipamento. Isso te agrada?";
	next;
	switch(select("Sim, agrada.:Quanto voce cobra?:Nao, obrigado.")) {
	case 1:
		mes "[Wise Old Woman]";
		mes "Muito bem. Qual item devo examinar para voce?";
		next;

		setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
		for( set .@i,1; .@i <= 10; set .@i,.@i+1 ) {
			if( getequipisequiped(.@indices[.@i]) )
				set .@menu$, .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			set .@menu$, .@menu$ + ":";
		}
		set .@part, .@indices[ select(.@menu$) ];
		if(!getequipisequiped(.@part)) {
			mes "[Wise Old Woman]";
			mes "Jovem... Voce nao esta usando nada ai de onde eu possa remover cartas.";
			close;
		}
		if(getequipcardcnt(.@part) == 0) {
			mes "[Wise Old Woman]";
			mes "Jovem... Nao ha cartas compostas neste item. Nao posso fazer nada com ele, infelizmente.";
			close;
		}
		set .@cardcount,getequipcardcnt(.@part);
		
		if (!checkweight(1202,(.@cardcount+1))) { 
			mes "^3355FFSo um momento!";
			mes "Eu nao posso oferecer";
			mes "nenhum dos meus servicos";
			mes "porque voce esta carregando";
			mes "coisas demais. Guarde seus";
			mes "itens extras no Armazem Kafra e volte~";
			close;
		}
		mes "[Wise Old Woman]";
		mes "Este item tem " + .@cardcount + " cartas compostas nele. Para realizar minha magia, vou precisar de " + (.zenycost+(.@cardcount * .percardcost)) + " zeny, um ^0000FFStar Crumb^000000 e uma ^0000FFYellow Gemstone^000000.";
		next;
		if(select("Muito bem. Faca isso.:Deixa pra la.") == 2) {
			mes "[Wise Old Woman]";
			mes "Muito bem. Volte quando precisar dos meus servicos.";
			close;
		}
		if((zeny < (.zenycost+(.@cardcount * .percardcost))) || (countitem(1000) < 1) || (countitem(715) < 1)) {
			mes "[Wise Old Woman]";
			mes "Voce nao tem todos os itens que preciso para fazer minha magia, crianca. Volte quando tiver.";
			close;
		}
		mes "[Wise Old Woman]";
		mes "Antes de comecar, devo te avisar--posso falhar. Se isso acontecer, posso destruir as cartas, o item, ou ambos. Nao dou reembolso. Dito isso, o que e mais importante pra voce: As cartas ou o item?";
		next;
		switch(select("Mudei de ideia.:O item.:As cartas.")) {
		case 1:
			mes "[Wise Old Woman]";
			mes "Muito bem. Volte quando precisar dos meus servicos.";
			close;
		case 2:
			set .@failtype,1;
			break;
		case 3:
			set .@failtype,2;
		}
		mes "[Wise Old Woman]";
		mes "Muito bem. Vou comecar.";
		set Zeny, Zeny - (.zenycost+(.@cardcount * .percardcost));
		delitem 1000,1; //Star_Crumb
		delitem 715,1; //Yellow_Gemstone
		
		// Replace the constants in the next 3 lines with failure chance values defined in refine_db.txt
		// First value = Total failure chance (item and cards destroyed)
		// Second value = Partial failure chance (one or the other is destroyed, player decides which one is safe)
		// Third value = Harmless failure chance (all that's lost is your investment)

		set .@failchance,rand(100);
		if (.faildestroy==1) {
			if(.@failchance < 2) {
				next;
				failedremovecards .@part,0;
				mes "[Wise Old Woman]";
				mes "O processo foi um fracasso total. Infelizmente o item e as cartas foram destruidos.";
				close;
			}

			if(.@failchance < 8) {
				if (.@failtype == 1) {
					next;
					failedremovecards .@part,1;
					mes "[Wise Old Woman]";
					mes "Embora eu tenha conseguido remover as cartas do item, elas foram destruidas no processo. O item, porem, esta bem.";
					close;
				}

				if (.@failtype == 2) {
					next;
					failedremovecards .@part,2;
					mes "[Wise Old Woman]";
					mes "Muito infeliz. Consegui remover as cartas, mas o item em si foi destruido no processo.";
					close;
				}
			}
		}

		if(.@failchance < 10) {
			next;
			failedremovecards .@part,3;
			mes "[Wise Old Woman]";
			mes "Falhei em remover as cartas. Felizmente, porem, tanto o item quanto as cartas ainda estao bem.";
			close;
		}
		next;
		successremovecards .@part;
		mes "[Wise Old Woman]";
		mes "O processo foi um sucesso. Aqui estao suas cartas e seu item. Ate mais.";
		close;
	case 2:
		mes "[Wise Old Woman]";
		mes "Eu cobro uma taxa fixa de "+callfunc("F_InsertComma",.zenycost)+" zeny, mais "+callfunc("F_InsertComma",.percardcost)+" zeny para cada carta que removo do item. Alem disso, preciso de um star crumb e uma yellow gemstone para fazer minha magia.";
		close;
	case 3:
		mes "[Wise Old Woman]";
		mes "Muito bem. Volte quando precisar dos meus servicos.";
		close;
	}
}
